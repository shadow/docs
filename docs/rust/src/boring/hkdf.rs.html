1//! HMAC-based Extract-and-Expand Key Derivation Function (HKDF) as specified by
2//! [RFC 5869](https://www.rfc-editor.org/rfc/rfc5869.html).
3
4use crate::error::ErrorStack;
5use crate::hash::{DigestBytes, MessageDigest};
6use crate::{cvt, ffi};
7
8/// HKDF for a given hash function.
9///
10/// The digest determines `HashLen`, the length of the pseudorandom key (PRK)
11/// returned by [`HkdfSuite::extract`]. HKDF-Expand limits output to
12/// `255 * HashLen` bytes, as specified by [RFC 5869, Section
13/// 2.3](https://www.rfc-editor.org/rfc/rfc5869.html#section-2.3).
14#[derive(Copy, Clone)]
15pub struct HkdfSuite {
16    digest: MessageDigest,
17}
18
19impl HkdfSuite {
20    /// Creates an HKDF suite using `digest`.
21    #[must_use]
22    pub fn new(digest: MessageDigest) -> Self {
23        Self { digest }
24    }
25
26    /// Returns `HashLen`, the digest output size in bytes.
27    #[must_use]
28    pub fn prk_size(&self) -> usize {
29        self.digest.size()
30    }
31
32    /// Computes HKDF-Extract as specified in [RFC 5869, Section
33    /// 2.2](https://www.rfc-editor.org/rfc/rfc5869.html#section-2.2).
34    ///
35    /// `salt` is an optional salt value. An empty `salt` is treated as an
36    /// omitted salt and is equivalent to a string of `HashLen` zero octets.
37    /// `ikm` is the input keying material. The returned pseudorandom key (PRK)
38    /// has length [`HkdfSuite::prk_size`].
39    pub fn extract(&self, salt: &[u8], ikm: &[u8]) -> Result<DigestBytes, ErrorStack> {
40        let mut prk = DigestBytes {
41            buf: [0u8; ffi::EVP_MAX_MD_SIZE as usize],
42            len: 0,
43        };
44        ffi::init();
45
46        // BoringSSL's native HKDF_extract API orders these arguments as
47        // secret/IKM, then salt, unlike the RFC 5869 notation and this API.
48        unsafe {
49            cvt(ffi::HKDF_extract(
50                prk.buf.as_mut_ptr(),
51                &mut prk.len,
52                self.digest.as_ptr(),
53                ikm.as_ptr(),
54                ikm.len(),
55                salt.as_ptr(),
56                salt.len(),
57            ))?;
58        }
59
60        debug_assert_eq!(prk.len, self.prk_size());
61
62        Ok(prk)
63    }
64
65    /// Computes HKDF-Expand as specified in [RFC 5869, Section
66    /// 2.3](https://www.rfc-editor.org/rfc/rfc5869.html#section-2.3).
67    ///
68    /// `prk` must be a pseudorandom key of at least `HashLen` bytes, usually
69    /// the output of [`HkdfSuite::extract`]. Shorter PRKs are rejected.
70    /// `info` is optional context and application-specific information and may
71    /// be empty. The function writes `L` bytes of output keying material into
72    /// `okm`, where `L` is its length. `okm` may be empty. HKDF limits `L` to
73    /// `255 * HashLen`; output-limit errors are returned as [`ErrorStack`].
74    pub fn expand(&self, prk: &[u8], info: &[u8], okm: &mut [u8]) -> Result<(), ErrorStack> {
75        if prk.len() < self.prk_size() {
76            return Err(ErrorStack::internal_error_str(
77                "HKDF PRK is shorter than the digest output",
78            ));
79        }
80
81        ffi::init();
82
83        unsafe {
84            cvt(ffi::HKDF_expand(
85                okm.as_mut_ptr(),
86                okm.len(),
87                self.digest.as_ptr(),
88                prk.as_ptr(),
89                prk.len(),
90                info.as_ptr(),
91                info.len(),
92            ))
93        }
94    }
95}
