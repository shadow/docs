22/// A router assists with moving packets between hosts across the simulated
23/// network.
24pub struct Router {
25    // Forwards packets out from our internet interface to the router.
26    relay_inet_out: Arc<Relay>,
27    // Forwards packets from the router in to our internet interface.
28    relay_inet_in: Arc<Relay>,
29    // Forwards packets from the localhost interface back to itself.
30    relay_loopback: Arc<Relay>,
31}
32
33impl Router {
34    /// Create a new router at the default address that will assist with moving
35    /// packets between interfaces and across the network.
36    pub fn new(addr: Ipv4Addr, limit_up: RateLimit, limit_down: RateLimit) -> Router {
37        Router {
38            relay_inet_out: Arc::new(Relay::new(limit_up, addr)),
39            relay_inet_in: Arc::new(Relay::new(limit_down, Ipv4Addr::UNSPECIFIED)),
40            relay_loopback: Arc::new(Relay::new(RateLimit::Unlimited, Ipv4Addr::LOCALHOST)),
41        }
42    }
43
44    pub fn notify_loopback(&self, host: &Host) {
45        self.relay_loopback.notify(host)
46    }
47
48    pub fn notify_default_out(&self, host: &Host) {
49        self.relay_inet_out.notify(host)
50    }
51
52    pub fn notify_default_in(&self, host: &Host) {
53        self.relay_inet_in.notify(host);
54    }
55}

57// Offers a device for handling packets coming in from the internet. Note: this
58// is a temporary workaround, and will be removed when the router redesign is
59// complete.
60pub struct RouterDevice {
61    magic: Magic<Self>,
62    _counter: ObjectCounter,
63    address: Ipv4Addr,
64    /// Packets inbound to the host from the simulated network.
65    inbound_packets: RefCell<CoDelQueue>,
66}
67
68impl RouterDevice {
69    /// Create a new router for a host that will help route packets between it
70    /// and other hosts. The `address` must uniquely identify this router to the
71    /// host that owns it.
72    pub fn new(address: Ipv4Addr) -> RouterDevice {
73        RouterDevice {
74            magic: Magic::new(),
75            address,
76            _counter: ObjectCounter::new("Router"),
77            inbound_packets: RefCell::new(CoDelQueue::default()),
78        }
79    }
80
81    fn push_inner(&self, packet: PacketRc, now: EmulatedTime) {
82        self.magic.debug_check();
83        self.inbound_packets.borrow_mut().push(packet, now);
84    }
85
86    fn pop_inner(&self, now: EmulatedTime) -> Option<PacketRc> {
87        self.magic.debug_check();
88        self.inbound_packets.borrow_mut().pop(now)
89    }
90
91    /// Routes the packet from the source host through the virtual internet to
92    /// the destination host.
93    fn route_outgoing_packet(&self, packet: PacketRc) {
94        // TODO: move Worker::send_packet to here?
95        Worker::with_active_host(|src_host| Worker::send_packet(src_host, packet)).unwrap();
96    }
97
98    /// Routes the packet from the virtual internet into our CoDel queue, which
99    /// can then be received by the destiantion host by calling pop().
100    pub fn route_incoming_packet(&self, packet: PacketRc) {
101        self.push_inner(packet, Worker::current_time().unwrap())
102    }
103}
104
105impl PacketDevice for RouterDevice {
106    fn get_address(&self) -> Ipv4Addr {
107        self.address
108    }
109
110    fn pop(&self) -> Option<PacketRc> {
111        // When the host calls pop, we provide the next packet from the CoDel queue.
112        self.pop_inner(Worker::current_time().unwrap())
113    }
114
115    fn push(&self, packet: PacketRc) {
116        // When the host calls push, we send to the virtual internet.
117        self.route_outgoing_packet(packet);
118    }
119}

121/// Configures a token bucket according the the given bytes_per_second rate
122/// limit. We always refill at least 1 byte per millisecond.
123fn create_token_bucket(bytes_per_second: u64) -> TokenBucket {
124    let refill_interval = SimulationTime::from_millis(1);
125    let refill_size = std::cmp::max(1, bytes_per_second / 1000);
126
127    // Only the `capacity` of the bucket is increased by the burst allowance,
128    // not the `refill_size`. Therefore, the long term rate limit enforced by
129    // the token bucket (configured by `refill_size`) is not affected much.
130    let capacity = refill_size + get_burst_allowance();
131
132    TokenBucket::new(capacity, refill_size, refill_interval).unwrap()
133}
134
135/// Returns the "burst allowance" we use in our token buckets.
136///
137/// What the burst allowance ensures is that we don't lose tokens that are
138/// unused because we don't fragment packets. If we set the capacity of the
139/// bucket to exactly the refill size (i.e., without the `CONFIG_MTU` burst
140/// allowance) and there are only 1499 tokens left in this sending round, a full
141/// packet would not fit. The next time the bucket refills, it adds
142/// `refill_size` tokens but in doing so 1499 tokens would fall over the top of
143/// the bucket; these tokens would represent wasted bandwidth, and could
144/// potentially accumulate in every refill interval leading to a significantly
145/// lower achievable bandwidth.
146///
147/// A downside of the `CONFIG_MTU` burst allowance is that the sending rate
148/// could possibly become "bursty" with a behavior such as:
149/// - interval 1: send `refill_size` + `CONFIG_MTU` bytes, sending over the
150///   allowance by 1500 bytes
151/// - refill: `refill_size` token gets added to the bucket
152/// - interval 2: send `refill_size` - `CONFIG_MTU` bytes, sending under the
153///   allowance by 1500 bytes
154/// - refill: `refill_size` token gets added to the bucket
155/// - interval 3: send `refill_size` + `CONFIG_MTU` bytes, sending over the
156///   allowance by 1500 bytes
157/// - repeat
158///
159/// So it could become less smooth and more "bursty" even though the long term
160/// average is maintained. But I don't think this would happen much in practice,
161/// and we are batching sends for performance reasons.
162fn get_burst_allowance() -> u64 {
163    cshadow::CONFIG_MTU.into()
164}
