76fn run(args: Args) -> Result<()> {
77    log::info!("Running the qperf check utility with args: {args:?}");
78
79    log::info!("Parsing sqlog file at {:?}", args.sqlog);
80
81    // First parse the log file and extract the metrics.
82    let records = QLogRecords::try_from(args.sqlog)?;
83
84    let mut metrics = Metrics::default();
85    for record in records.into_iter() {
86        metrics.append(&mut Metrics::try_from(record)?);
87    }
88
89    log::info!("Found {} total metrics", metrics.len());
90
91    // Now compute some high-level stats.
92    let events = metrics.len();
93    let packets_sent = metrics.packets_sent();
94    let packets_received = metrics.packets_received();
95    let packets_lost = metrics.packets_lost();
96    let packets_retransmitted = metrics.packets_retransmitted();
97    let network_bytes_sent = metrics.network_bytes_sent();
98    let network_bytes_received = metrics.network_bytes_received();
99    let stream_bytes_sent = metrics.stream_bytes_sent();
100    let stream_bytes_received = metrics.stream_bytes_received();
101
102    let rtt_ms_min = metrics.rtt_quantile(0.0);
103    let rtt_ms_p25 = metrics.rtt_quantile(0.25);
104    let rtt_ms_p50 = metrics.rtt_quantile(0.5);
105    let rtt_ms_p75 = metrics.rtt_quantile(0.75);
106    let rtt_ms_p95 = metrics.rtt_quantile(0.95);
107    let rtt_ms_max = metrics.rtt_quantile(1.0);
108
109    let rtt_variance_ms_min = metrics.rtt_variance_quantile(0.0);
110    let rtt_variance_ms_p25 = metrics.rtt_variance_quantile(0.25);
111    let rtt_variance_ms_p50 = metrics.rtt_variance_quantile(0.5);
112    let rtt_variance_ms_p75 = metrics.rtt_variance_quantile(0.75);
113    let rtt_variance_ms_p95 = metrics.rtt_variance_quantile(0.95);
114    let rtt_variance_ms_max = metrics.rtt_variance_quantile(1.0);
115
116    let rtt_smoothed_ms_min = metrics.rtt_smoothed_quantile(0.0);
117    let rtt_smoothed_ms_p25 = metrics.rtt_smoothed_quantile(0.25);
118    let rtt_smoothed_ms_p50 = metrics.rtt_smoothed_quantile(0.5);
119    let rtt_smoothed_ms_p75 = metrics.rtt_smoothed_quantile(0.75);
120    let rtt_smoothed_ms_p95 = metrics.rtt_smoothed_quantile(0.95);
121    let rtt_smoothed_ms_max = metrics.rtt_smoothed_quantile(1.0);
122
123    let pto_max = metrics.pto_max();
124    let congestion_window_max = metrics.congestion_window_max();
125    let bytes_in_flight_max = metrics.bytes_in_flight_max();
126    let packets_in_flight_max = metrics.packets_in_flight_max();
127
128    let ss_threshold_min = metrics.ss_threshold_quantile(0.0);
129    let ss_threshold_p50 = metrics.ss_threshold_quantile(0.5);
130    let ss_threshold_max = metrics.ss_threshold_quantile(1.0);
131
132    let loss_percent = metrics.loss_percent();
133    let transfer_duration_ms = metrics.transfer_duration_ms();
134    let goodput_mbps_up = metrics.goodput_mbps_up();
135    let goodput_mbps_down = metrics.goodput_mbps_down();
136    let throughput_mbps_up = metrics.throughput_mbps_up();
137    let throughput_mbps_down = metrics.throughput_mbps_down();
138
139    // Check that some stats are within the configured tolerances.
140    let mut failures = Vec::<String>::new();
141
142    // Minimum rount-trip time should be within a tolerance of the expected minimum RTT.
143    if let Some(expected) = args.rtt_min_ms
144        && let Some(actual) = rtt_ms_min
145        && (actual - expected).abs() > args.rtt_min_ms_tol
146    {
147        let msg = format!(
148            "Minimum RTT {actual:.2} is outside {expected:.2} +/- {:.2}",
149            args.rtt_min_ms_tol
150        );
151        failures.push(msg);
152    }
153
154    // Median rount-trip time should be within a tolerance of the expected median RTT.
155    if let Some(expected) = args.rtt_med_ms
156        && let Some(actual) = rtt_ms_p50
157        && (actual - expected).abs() > args.rtt_med_ms_tol
158    {
159        let msg = format!(
160            "Median RTT {actual:.2} is outside {expected:.2} +/- {:.2}",
161            args.rtt_med_ms_tol
162        );
163        failures.push(msg);
164    }
165
166    // Packet loss should be within a tolerance of the expected loss.
167    if let Some(expected) = args.loss_percent
168        && let Some(actual) = loss_percent
169        && (actual - expected).abs() > args.loss_percent_tol
170    {
171        let msg = format!(
172            "loss {actual:.2}% is outside {expected:.2}% +/- {:.2}%",
173            args.loss_percent_tol
174        );
175        failures.push(msg);
176    }
177
178    // Transfer of packets and application bytes.
179    if packets_sent == 0 {
180        failures.push("no packet_sent events found".to_string());
181    }
182    if packets_received == 0 {
183        failures.push("no packets_received events found".to_string());
184    }
185    if stream_bytes_sent == 0 {
186        failures.push("no stream_bytes_sent events found".to_string());
187    }
188    if stream_bytes_received == 0 {
189        failures.push("no stream_bytes_received events found".to_string());
190    }
191
192    // Goodput should be within a tolerance of expected bandwidth.
193    if stream_bytes_sent > args.min_goodput_bytes
194        && let Some(expected) = args.bw_up_mbps
195        && let Some(actual) = goodput_mbps_up
196    {
197        check_goodput(expected, actual, args.bw_percent_tol, "up", &mut failures);
198    }
199    if stream_bytes_received > args.min_goodput_bytes
200        && let Some(expected) = args.bw_down_mbps
201        && let Some(actual) = goodput_mbps_down
202    {
203        check_goodput(expected, actual, args.bw_percent_tol, "down", &mut failures);
204    }
205
206    // In flight bytes should not exceed congestion window.
207    if let Some(cwnd_max) = congestion_window_max
208        && let Some(flight_max) = bytes_in_flight_max
209    {
210        let tol = args.cwnd_flight_tol.clamp(0.0, 100.0);
211        let threshold = (cwnd_max as f64) * (1.0 + (tol / 100.0));
212
213        if flight_max as f64 > threshold {
214            let msg = format!(
215                "bytes_in_flight {flight_max} exceeds congestion_window {cwnd_max} \
216                by more than {:.2}%",
217                args.cwnd_flight_tol
218            );
219            failures.push(msg);
220        }
221    }
222
223    // Write out the results.
224    let output = serde_json::json!({
225        "events": events,
226        "packets_sent": packets_sent,
227        "packets_received": packets_received,
228        "packets_lost": packets_lost,
229        "packets_retransmitted": packets_retransmitted,
230        "network_bytes_sent": network_bytes_sent,
231        "network_bytes_received": network_bytes_received,
232        "stream_bytes_sent": stream_bytes_sent,
233        "stream_bytes_received": stream_bytes_received,
234        "rtt_ms_min": rtt_ms_min,
235        "rtt_ms_p25": rtt_ms_p25,
236        "rtt_ms_p50": rtt_ms_p50,
237        "rtt_ms_p75": rtt_ms_p75,
238        "rtt_ms_p95": rtt_ms_p95,
239        "rtt_ms_max": rtt_ms_max,
240        "rtt_variance_ms_min": rtt_variance_ms_min,
241        "rtt_variance_ms_p25": rtt_variance_ms_p25,
242        "rtt_variance_ms_p50": rtt_variance_ms_p50,
243        "rtt_variance_ms_p75": rtt_variance_ms_p75,
244        "rtt_variance_ms_p95": rtt_variance_ms_p95,
245        "rtt_variance_ms_max": rtt_variance_ms_max,
246        "rtt_smoothed_ms_min": rtt_smoothed_ms_min,
247        "rtt_smoothed_ms_p25": rtt_smoothed_ms_p25,
248        "rtt_smoothed_ms_p50": rtt_smoothed_ms_p50,
249        "rtt_smoothed_ms_p75": rtt_smoothed_ms_p75,
250        "rtt_smoothed_ms_p95": rtt_smoothed_ms_p95,
251        "rtt_smoothed_ms_max": rtt_smoothed_ms_max,
252        "pto_max": pto_max,
253        "congestion_window_max": congestion_window_max,
254        "bytes_in_flight_max": bytes_in_flight_max,
255        "packets_in_flight_max": packets_in_flight_max,
256        "ss_threshold_min": ss_threshold_min,
257        "ss_threshold_p50": ss_threshold_p50,
258        "ss_threshold_max": ss_threshold_max,
259        "loss_percent": loss_percent,
260        "transfer_duration_ms": transfer_duration_ms,
261        "goodput_mbps_up": goodput_mbps_up,
262        "goodput_mbps_down": goodput_mbps_down,
263        "throughput_mbps_up": throughput_mbps_up,
264        "throughput_mbps_down": throughput_mbps_down,
265        "failures": failures,
266    });
267
268    eprintln!("{}", serde_json::to_string_pretty(&output)?);
269
270    log::info!("Done! Summary metrics were written to stderr.");
271
272    if failures.is_empty() {
273        Ok(())
274    } else {
275        bail!(
276            "{} metrics checks failed (see stderr for details)",
277            failures.len()
278        )
279    }
280}

282fn check_goodput(
283    expected: f64,
284    actual: f64,
285    percent_tol: f64,
286    direction: &str,
287    failures: &mut Vec<String>,
288) {
289    if actual > expected {
290        let msg = format!(
291            "goodput {direction} {actual:.2} Mbit/s exceeded \
292            the theoretical maximum {expected:.2} Mbit/s"
293        );
294        failures.push(msg);
295    }
296
297    let tol = percent_tol.clamp(0.0, 100.0);
298    let threshold = (1.0 - (tol / 100.0)) * expected;
299
300    if actual < threshold {
301        let msg = format!(
302            "goodput {direction} {actual:.2} Mbit/s was less than \
303            the allowed minimum threshold {threshold:.2} Mbit/s"
304        );
305        failures.push(msg);
306    }
307}

309#[derive(Debug)]
310struct QLogRecords {
311    records: Vec<Record>,
312}
313
314impl TryFrom<PathBuf> for QLogRecords {
315    type Error = anyhow::Error;
316
317    fn try_from(path: PathBuf) -> Result<Self> {
318        // Get the file contents.
319        let contents = fs::read(&path).context(format!("failed to read {}", path.display()))?;
320
321        // Store the records parsed from the file.
322        let mut records = Vec::new();
323        let mut n_rows = 0usize;
324
325        // The file is in the JSON text sequence format (json-seq). Each record is
326        // prefixed by an ASCII Record Separator (0x1E) and ends with an ASCII Line
327        // Feed character (0x0A): https://datatracker.ietf.org/doc/rfc7464/
328        //
329        // Skip line 1: it's empty as an artifact of this split method.
330        for record in contents.split(|&b| b == 0x1E).skip(1) {
331            n_rows += 1;
332
333            // Trim whitespace and the line feed.
334            let trimmed = record.trim_ascii();
335
336            if trimmed.is_empty() {
337                log::warn!("Skipping empty record on row {n_rows}");
338                continue;
339            }
340
341            // Get the serde value, and then convert to our typed record.
342            match serde_json::from_slice::<Value>(trimmed) {
343                Ok(value) => match Record::try_from(value) {
344                    Ok(event) => records.push(event),
345                    Err(e) => log::warn!("Skipping invalid JSON event on row {n_rows}: {e}"),
346                },
347                Err(e) => log::warn!("Skipping invalid JSON valueon row {n_rows}: {e}"),
348            }
349        }
350
351        log::info!(
352            "Found {} valid records and skipped {} invalid records",
353            records.len(),
354            n_rows - records.len()
355        );
356
357        Ok(Self { records })
358    }
359}

367#[derive(Debug)]
368enum Record {
369    // We do not yet process the header.
370    #[allow(dead_code)]
371    Header(Header),
372    Event(Event),
373}
374
375impl TryFrom<Value> for Record {
376    type Error = anyhow::Error;
377
378    fn try_from(value: Value) -> Result<Self> {
379        let object = value
380            .as_object()
381            .context("failed to get object from JSON value")?;
382
383        let record = if object.contains_key("file_schema") {
384            Record::Header(Header::try_from(object.clone())?)
385        } else {
386            Record::Event(Event::try_from_ref(object)?)
387        };
388
389        Ok(record)
390    }
391}

393#[derive(Debug)]
394struct Header {
395    _data: Map<String, Value>,
396}
397
398impl TryFrom<Map<String, Value>> for Header {
399    type Error = anyhow::Error;
400
401    fn try_from(object: Map<String, Value>) -> Result<Self> {
402        // Event format: {"file_schema": ..., ...}
403        Ok(Self { _data: object })
404    }
405}

407#[derive(Debug)]
408struct Event {
409    time: f64,
410    name: String,
411    data: Map<String, Value>,
412}
413
414impl Event {
415    // Like TryFrom, but with an object ref so we can avoid a clone.
416    fn try_from_ref(object: &Map<String, Value>) -> Result<Self> {
417        // Event format: {"time": ..., "name": ..., "data": {...}}
418        let time = object
419            .get("time")
420            .context("missing the 'time' key in JSON object")?
421            .as_f64()
422            .context("incorrect type for 'time' value")?;
423
424        let name = object
425            .get("name")
426            .context("missing the 'name' key in JSON object")?
427            .as_str()
428            .context("incorrect type for 'name' value")?
429            .to_string();
430
431        let data = object
432            .get("data")
433            .context("missing the 'data' key in JSON object")?
434            .as_object()
435            .context("incorrect type for 'data' value")?
436            .clone();
437
438        Ok(Event { time, name, data })
439    }
440
441    fn time(&self) -> Duration {
442        // sqlog records start at t=0;
443        Duration::from_secs_f64(self.time / 1_000.0)
444    }
445
446    // Get the raw packet length of the packet.
447    fn get_raw_length(&self) -> Option<u64> {
448        self.data
449            .get("raw")
450            .and_then(|raw| raw.get("length").and_then(Value::as_u64))
451    }
452
453    // Get the total payload length across all frames in the packet.
454    fn get_frames_raw_payload_length(&self) -> Option<u64> {
455        let frames = self.data.get("frames")?.as_array()?;
456
457        let total_payload_len = frames
458            .iter()
459            .filter_map(|frame| {
460                frame
461                    .get("raw")
462                    .and_then(|raw| raw.get("payload_length").and_then(Value::as_u64))
463            })
464            .sum();
465
466        Some(total_payload_len)
467    }
468
469    fn get_f64(&self, key: &str) -> Option<f64> {
470        self.data.get(key).and_then(Value::as_f64)
471    }
472
473    fn get_u64(&self, key: &str) -> Option<u64> {
474        self.data.get(key).and_then(Value::as_u64)
475    }
476
477    fn get_string(&self, key: &str) -> Option<String> {
478        self.data.get(key).and_then(Value::as_str).map(String::from)
479    }
480}

512#[derive(Debug, Default)]
513struct Metrics {
514    metrics: Vec<Metric>,
515}
516
517impl Metrics {
518    fn append(&mut self, other: &mut Self) {
519        self.metrics.append(&mut other.metrics);
520    }
521
522    fn len(&self) -> usize {
523        self.metrics.len()
524    }
525
526    fn sum_by<T, F>(&self, extract: F) -> T
527    where
528        T: std::iter::Sum,
529        F: Fn(&Metric) -> Option<T>,
530    {
531        self.metrics.iter().filter_map(extract).sum()
532    }
533
534    fn min_by<T, F>(&self, extract: F) -> Option<T>
535    where
536        T: Ord,
537        F: Fn(&Metric) -> Option<T>,
538    {
539        self.metrics.iter().filter_map(extract).min()
540    }
541
542    fn max_by<T, F>(&self, extract: F) -> Option<T>
543    where
544        T: Ord,
545        F: Fn(&Metric) -> Option<T>,
546    {
547        self.metrics.iter().filter_map(extract).max()
548    }
549
550    fn _min_f64_by<F>(&self, extract: F) -> Option<f64>
551    where
552        F: Fn(&Metric) -> Option<f64>,
553    {
554        self.metrics
555            .iter()
556            .filter_map(extract)
557            .min_by(f64::total_cmp)
558    }
559
560    fn _max_f64_by<F>(&self, extract: F) -> Option<f64>
561    where
562        F: Fn(&Metric) -> Option<f64>,
563    {
564        self.metrics
565            .iter()
566            .filter_map(extract)
567            .max_by(f64::total_cmp)
568    }
569
570    /// Computes the given quantile 'q' (e.g., 0.95 for 95th percentile).
571    fn quantile_by<T, F>(&self, q: f64, extract: F) -> Option<T>
572    where
573        T: Clone + PartialOrd,
574        F: Fn(&Metric) -> Option<T>,
575    {
576        // Collect all matching metric values.
577        let mut values: Vec<T> = self.metrics.iter().filter_map(extract).collect();
578
579        if values.is_empty() {
580            return None;
581        }
582
583        // Sort the values in ascending order. Support floating-point values
584        // with `partial_cmp`.
585        values.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
586
587        // Calculate the target index using the Nearest Rank method.
588        let max_index = values.len() - 1;
589        let q_index = q * (max_index as f64);
590        let index = (q_index.round() as usize).clamp(0, max_index);
591
592        // Return the value at that rank.
593        Some(values[index].clone())
594    }
595
596    fn packets_sent(&self) -> usize {
597        self.sum_by(|m| match m {
598            Metric::PacketSent => Some(1),
599            _ => None,
600        })
601    }
602
603    fn packets_received(&self) -> usize {
604        self.sum_by(|m| match m {
605            Metric::PacketReceived => Some(1),
606            _ => None,
607        })
608    }
609
610    fn packets_lost(&self) -> usize {
611        self.sum_by(|m| match m {
612            Metric::PacketLost => Some(1),
613            _ => None,
614        })
615    }
616
617    fn packets_retransmitted(&self) -> usize {
618        self.sum_by(|m| match m {
619            Metric::PacketRetransmitted => Some(1),
620            _ => None,
621        })
622    }
623
624    fn network_bytes_sent(&self) -> u64 {
625        self.sum_by(|m| match m {
626            Metric::NetworkBytesSent(b, _) => Some(*b),
627            _ => None,
628        })
629    }
630
631    fn network_bytes_received(&self) -> u64 {
632        self.sum_by(|m| match m {
633            Metric::NetworkBytesReceived(b, _) => Some(*b),
634            _ => None,
635        })
636    }
637
638    fn stream_bytes_sent(&self) -> u64 {
639        self.sum_by(|m| match m {
640            Metric::StreamBytesSent(b, _) => Some(*b),
641            _ => None,
642        })
643    }
644
645    fn stream_bytes_received(&self) -> u64 {
646        self.sum_by(|m| match m {
647            Metric::StreamBytesReceived(b, _) => Some(*b),
648            _ => None,
649        })
650    }
651
652    fn rtt_quantile(&self, q: f64) -> Option<f64> {
653        self.quantile_by(q, |m| match m {
654            Metric::RttLatest(t) => Some(*t),
655            _ => None,
656        })

708    fn loss_percent(&self) -> Option<f64> {
709        let sent = self.packets_sent();
710        if sent > 0 {
711            let loss = self.packets_lost();
712            Some((loss as f64) * 100.0 / (sent as f64))

717
718    fn transfer_duration_ms(&self) -> f64 {
719        let filter = |m: &Metric| match m {
720            Metric::StreamBytesSent(_, t) => Some(*t),
721            Metric::StreamBytesReceived(_, t) => Some(*t),
722            _ => None,
723        };
724
725        let first = self.min_by(filter);
726        let last = self.max_by(filter);
727
728        if let Some(f) = first
729            && let Some(l) = last
730        {
731            (l - f).as_secs_f64() * 1_000.0
732        } else {
733            0.0
734        }
735    }
736
737    fn mbit_per_sec_by<F>(&self, extract: F) -> Option<f64>
738    where
739        F: Fn(&Metric) -> Option<(u64, Duration)>,
740    {
741        let mut first: Option<Duration> = None;
742        let mut last: Option<Duration> = None;
743        let mut bytes: usize = 0;
744
745        for metric in self.metrics.iter() {
746            if let Some((b, t)) = extract(metric) {
747                // Keep a sum of the bytes.
748                bytes += b as usize;
749
750                // Keep track of the min time.
751                first = Some(first.map_or(t, |f| f.min(t)));
752
753                // Keep track of the last time.
754                last = Some(last.map_or(t, |f| f.max(t)));
755            }
756        }
757
758        let (Some(first), Some(last)) = (first, last) else {
759            return None;
760        };
761
762        let duration_secs = (last - first).as_secs_f64();
763        if duration_secs == 0.0 {
764            return None;
765        }
766
767        let mbit = (bytes * 8) as f64 / 1_000_000.0;
768
769        Some(mbit / duration_secs)
770    }

772    fn goodput_mbps_up(&self) -> Option<f64> {

775            _ => None,
776        })
777    }

801impl TryFrom<Record> for Metrics {
802    type Error = anyhow::Error;
803
804    fn try_from(value: Record) -> Result<Self> {
805        match value {
806            Record::Header(_) => Ok(Metrics::default()), // Header has no metrics.
807            Record::Event(event) => Metrics::try_from(event),
808        }
809    }
810}
811
812impl TryFrom<Event> for Metrics {
813    type Error = anyhow::Error;
814
815    fn try_from(event: Event) -> Result<Self> {
816        let mut metrics = Vec::new();
817
818        // Example data layout for quic:packet_* event types.
819        // "data":{
820        //    "header":{"packet_type":"1RTT","packet_number":13},
821        //    "raw":{"length":1200,"payload_length":1166},
822        //    "send_at_time":200.0,
823        //    "frames":[
824        //       {"frame_type":"stream","stream_id":0,"offset":9114,"raw":{"payload_length":1160}}
825        //    ]
826        // }
827
828        match event.name.as_str() {
829            "quic:packet_sent" => {
830                metrics.push(Metric::PacketSent);
831                if let Some(len) = event.get_raw_length() {
832                    metrics.push(Metric::NetworkBytesSent(len, event.time()))
833                }
834                if let Some(len) = event.get_frames_raw_payload_length() {
835                    metrics.push(Metric::StreamBytesSent(len, event.time()))
836                }
837            }
838
839            "quic:packet_received" => {
840                metrics.push(Metric::PacketReceived);
841                if let Some(len) = event.get_raw_length() {
842                    metrics.push(Metric::NetworkBytesReceived(len, event.time()))
843                }
844                if let Some(len) = event.get_frames_raw_payload_length() {
845                    metrics.push(Metric::StreamBytesReceived(len, event.time()))
846                }
847            }
848
849            "quic:packet_lost" => {
850                metrics.push(Metric::PacketLost);
851            }
852
853            "quic:packet_retransmitted" => {
854                metrics.push(Metric::PacketRetransmitted);
855            }
856
857            "quic:recovery_metrics_updated" => {
858                if let Some(val) = event.get_f64("min_rtt") {
859                    metrics.push(Metric::RttMin(val));
860                }
861                if let Some(val) = event.get_f64("min_rtt") {
862                    metrics.push(Metric::RttLatest(val));
863                }
864                if let Some(val) = event.get_f64("smoothed_rtt") {
865                    metrics.push(Metric::RttSmoothed(val));
866                }
867                if let Some(val) = event.get_f64("rtt_variance") {
868                    metrics.push(Metric::RttVariance(val));
869                }
870                if let Some(val) = event.get_u64("congestion_window") {
871                    metrics.push(Metric::CongestionWindow(val));
872                }
873                if let Some(val) = event.get_u64("ssthresh") {
874                    metrics.push(Metric::SsThresh(val));
875                }
876                if let Some(val) = event.get_u64("bytes_in_flight") {
877                    metrics.push(Metric::BytesInFlight(val));
878                }
879                if let Some(val) = event.get_u64("packets_in_flight") {
880                    metrics.push(Metric::PacketsInFlight(val));
881                }
882                if let Some(val) = event.get_u64("pto_count") {
883                    metrics.push(Metric::ProbeTimeout(val));
884                }
885            }
886
887            "quic:congestion_state_updated" => {
888                if let Some(val) = event.get_string("new") {
889                    metrics.push(Metric::CongestionState(val));
890                }
891            }
892
893            _ => metrics.push(Metric::Unidentified(event.name.clone())),
894        }
895
896        Ok(Self { metrics })
897    }
