7/// Runs a quic-based performance test between a quic client and a quic server.
8#[derive(Parser)]

11    #[command(subcommand)]

15/// The perf test can be run in either client mode or server mode.
16#[derive(Subcommand)]
17pub enum Command {
18    Client(ClientArgs),
19    Server(ServerArgs),
20}

22#[derive(Args)]
23pub struct ClientArgs {
24    #[command(flatten)]
25    pub common: CommonArgs,
26    /// The address of the quic perf server to which we send messages.

31#[derive(Args)]
32pub struct ServerArgs {

37#[derive(Debug, Args)]
38pub struct CommonArgs {
39    /// The address to bind to when receiving messages (use port 0 to auto-select).
40    #[arg(short, long, value_name = "ADDR:PORT", default_value = "0.0.0.0:0")]
41    pub bind: SocketAddr,
42    /// Number of bytes to send to the quic peer.
43    #[arg(short, long, value_name = "N", default_value = "1024")]
44    pub num_bytes: usize,
45    /// Path to store the QUIC performance metrics in standard sqlog format.
46    #[arg(
47        short,
48        long,
49        value_name = "PATH",

53}

55/// Create a new quic config. Both the client and server should use this to
56/// ensure protocol alignment.
57pub fn init_quic_config() -> anyhow::Result<Config> {
58    // Synchronize the client and server protocol versions.
59    let mut config = Config::new(quiche::PROTOCOL_VERSION)?;
60    config.set_application_protos(&[b"quic-perf-test-shadow"])?;
61
62    // Bypass verification checks to allow self-signed certificates.
63    config.verify_peer(false);
64
65    // Set the congestion control algorithm.
66    config.set_cc_algorithm(quiche::CongestionControlAlgorithm::CUBIC);
67
68    // Allow stream data to be transferred.
69    config.set_initial_max_streams_bidi(1);
70    config.set_initial_max_streams_uni(1);
71    config.set_initial_max_data(1_000_000);
72    config.set_initial_max_stream_data_bidi_local(1_000_000);
73    config.set_initial_max_stream_data_bidi_remote(1_000_000);
74    config.set_initial_max_stream_data_uni(1_000_000);
75
76    // No need to support session resumption.
77    config.set_disable_active_migration(true);
78
79    // After our transfer finishes, close the idle connection.
80    config.set_max_idle_timeout(1_000);
81
82    Ok(config)
83}

85/// Sets quic to store qlogs in the path specified in the env var `QLOGDIR` or
86/// else in the local directory if `QLOGDIR` is not set.
87pub fn enable_quic_qlog(
88    conn: &mut Connection,
89    sqlog_path: PathBuf,
90    role: &str,
91) -> std::io::Result<()> {
92    // Create the file.
93    let file = std::fs::File::create(sqlog_path)?;
94
95    // Configure the connection to store performance stats in the file.
96    conn.set_qlog(
97        Box::new(file),
98        format!("{role}-trace"),
99        format!("Quic {role} perf test"),
100    );
101
102    Ok(())
103}
