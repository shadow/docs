1//! Stream manager for a QUIC-based client/server performance tester.
2//!
3//! This module implements a tiny state machine that drives a single QUIC
4//! stream (client-initiated or server-discovered) through the phases required
5//! to complete a QUIC-based payload transfer:
6//!
7//! 1. Handshake - wait for the QUIC connection to be established.
8//! 2. Identify - determine the stream identifier.
9//! 3. Read/Write - exchange payload data.
10//! 4. Close - ensure the stream is finished.
11//! 5. Finished - all incoming stream data has been, and outgoing data pushed to
12//!    the QUIC connection buffers.
13
14use quiche::Connection;
15
16const CHUNK_SIZE: usize = 8192;
17
18/// Defines the progress of the stream transfer process
19/// (`Handshake -> Identify -> Read/Write -> Close -> Finished`).
20#[derive(Debug, Clone, Copy, PartialEq, Eq)]
21pub enum StreamState {
22    /// Connection is still being established.
23    Handshake,
24    /// Stream identifier is being negotiated (client already knows it, server
25    /// discovers it from the first readable stream).
26    Identify,
27    /// Both reading and writing are enabled.
28    ReadWrite,
29    /// Writing has completed and we are only reading data from the peer.
30    Read,
31    /// Reading has completed and we are only writing data to the peer.
32    Write,
33    /// Waiting for verify that the QUIC connection considers our stream closed.
34    Close,
35    /// All stream I/O is complete and the stream is closed.
36    Finished,
37}
38
39/// A manager used to drive a single QUIC stream for a data-transfer test.
40pub struct StreamManager {
41    /// The QUIC stream identifier. `None` for a server until the first readable
42    /// stream is observed.
43    stream_id: Option<u64>,
44    /// Temporary static, fixed-size buffer used for both stream reads and writes.
45    buf: Vec<u8>,
46    /// Bytes left to send to the peer.
47    remaining: usize,
48    /// Current state in our state machine.
49    state: StreamState,
50}
51
52impl StreamManager {
53    /// Create a client-side manager with a predefined stream ID (`0`).
54    ///
55    /// The client knows the stream identifier from the outset, so the manager
56    /// is constructed with `stream_id = Some(0)`.  The caller supplies the
57    /// total number of payload bytes to be transmitted.
58    pub fn new_client(num_bytes: usize) -> Self {
59        Self::new(num_bytes, Some(0))
60    }
61
62    /// Create a server-side manager that will discover the stream ID.
63    ///
64    /// The server starts with `stream_id = None` and will set it once a
65    /// readable stream appears on the connection.
66    pub fn new_server(num_bytes: usize) -> Self {
67        Self::new(num_bytes, None)
68    }
69
70    /// Core constructor shared by the client and server helpers. Allocates a
71    /// fixed-size buffer and initializes the state machine to
72    /// [`StreamState::Handshake`].
73    fn new(num_bytes: usize, stream_id: Option<u64>) -> Self {
74        Self {
75            stream_id,
76            buf: vec![0u8; CHUNK_SIZE],
77            remaining: num_bytes,
78            state: StreamState::Handshake,
79        }
80    }
81
82    /// Returns `true` iff the manager has reached [`StreamState::Finished`].
83    pub fn finished(&self) -> bool {
84        self.state == StreamState::Finished
85    }
86
87    /// Transition to a new `StreamState` and emit a log line.
88    ///
89    /// This helper centralises state changes so that the transition is always
90    /// recorded with `log::info!`.
91    fn set_state(&mut self, state: StreamState) {
92        log::info!("Transitioning from {:?} to {:?}", self.state, state);
93        self.state = state;
94    }
95
96    /// Processes all incoming and outgoing application data for the stream.
97    /// Returns the updated `StreamStatus`.
98    ///
99    /// Drive the stream forward by handling all pending reads and writes.
100    ///
101    /// The method should be called repeatedly by the I/O main loop. It examines
102    /// [`Self::state`], interacts with the provided [`quiche::Connection`], and
103    /// returns the **new** [`StreamState`] or an error propagated from the QUIC
104    /// library.
105    ///
106    /// The state-machine logic is:
107    ///
108    /// 1. **Handshake -> Identify** - once the connection is established.
109    /// 2. **Identify -> ReadWrite** - client already knows the ID; server gets
110    ///    it from the first readable stream on the connection.
111    /// 3. **ReadWrite / Read** - pull inbound stream data from the connection.
112    ///    When we read a `fin`, transition to [`StreamState::Close`], or
113    ///    [`StreamState::Write`] if we have yet to finish writing all stream
114    ///    data.
115    /// 4. **ReadWrite / Write** - push outbound stream data to the connection,
116    ///    respecting QUIC flow- and congestion-control. When all bytes and the
117    ///    `fin` are sent, transition to [`StreamState::Close`], or
118    ///    [`StreamState::Read`] if we have yet to finish reading all stream
119    ///    data.
120    /// 5. **Close** - transition to [`StreamState::Finished`] when quiche
121    ///    indicates the stream is finished and closed.
122    pub fn process(&mut self, conn: &mut Connection) -> Result<StreamState, quiche::Error> {
123        if self.state == StreamState::Handshake && conn.is_established() {
124            self.set_state(StreamState::Identify);
125        }
126
127        if self.state == StreamState::Identify {
128            // Client already has the id, server discovers it.
129            if let Some(sid) = self.stream_id {
130                log::info!("Client initiating stream {sid}");
131                self.set_state(StreamState::ReadWrite);
132            } else if let Some(sid) = conn.readable().next() {
133                log::info!("Server discovered new stream {sid}");
134                self.stream_id = Some(sid);
135                self.set_state(StreamState::ReadWrite);
136            }
137        }
138
139        // Read loop to drain the connection of all stream payload data.
140        if matches!(self.state, StreamState::ReadWrite | StreamState::Read) {
141            let sid = *self.stream_id.as_ref().unwrap();
142
143            while let Ok((len, fin)) = conn.stream_recv(sid, &mut self.buf) {
144                log::debug!("Stream {sid} dequeued {len} bytes. fin={fin}");
145
146                if fin {
147                    // Reading is done.
148                    let next = match self.state {
149                        StreamState::ReadWrite => StreamState::Write,
150                        StreamState::Read => StreamState::Close,
151                        _ => unreachable!(),
152                    };
153                    self.set_state(next);
154                }
155            }
156        }
157
158        // Write stream data as fast as window flow-control allows.
159        if matches!(self.state, StreamState::ReadWrite | StreamState::Write) {
160            let sid = *self.stream_id.as_ref().unwrap();
161
162            while self.remaining > 0 {
163                let to_write = std::cmp::min(self.remaining, CHUNK_SIZE);
164                let fin = self.remaining == to_write;
165
166                match conn.stream_send(sid, &self.buf[..to_write], fin) {
167                    Ok(len) => {
168                        self.remaining = self.remaining.saturating_sub(len);
169                        log::debug!(
170                            "Stream {sid} enqueued {len} bytes. Remaining: {}, fin={fin}",
171                            self.remaining
172                        );
173
174                        if self.remaining == 0 {
175                            // Writing is done.
176                            let next = match self.state {
177                                StreamState::ReadWrite => StreamState::Read,
178                                StreamState::Write => StreamState::Close,
179                                _ => unreachable!(),
180                            };
181                            self.set_state(next);
182                        }
183                    }
184                    Err(quiche::Error::Done) => break,
185                    Err(e) => return Err(e),
186                }
187            }
188        }
189
190        // If we are in the `Close` state, we should have already gotten a read
191        // fin, and should have indicated to quiche of our write fin. So
192        // effectively, quiche likely already considers us closed.
193        // TODO: maybe we can remove the `Close` state, and this block.
194        if self.state == StreamState::Close {
195            let sid = *self.stream_id.as_ref().unwrap();
196            if conn.stream_finished(sid) && conn.stream_closed(sid) {
197                self.set_state(StreamState::Finished);
198            }
199        }
200
201        Ok(self.state)
202    }
203}
