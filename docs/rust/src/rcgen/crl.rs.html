16/// A certificate revocation list (CRL)
17///
18/// ## Example
19///
20/// ```
21/// extern crate rcgen;
22/// use rcgen::*;
23///
24/// #[cfg(not(feature = "crypto"))]
25/// struct MyKeyPair { public_key: Vec<u8> }
26/// #[cfg(not(feature = "crypto"))]
27/// impl SigningKey for MyKeyPair {
28///   fn sign(&self, _: &[u8]) -> Result<Vec<u8>, rcgen::Error> { Ok(vec![]) }
29/// }
30/// #[cfg(not(feature = "crypto"))]
31/// impl PublicKeyData for MyKeyPair {
32///   fn der_bytes(&self) -> &[u8] { &self.public_key }
33///   fn algorithm(&self) -> &'static SignatureAlgorithm { &PKCS_ED25519 }
34/// }
35/// # fn main () {
36/// // Generate a CRL issuer.
37/// let mut issuer_params = CertificateParams::new(vec!["crl.issuer.example.com".to_string()]).unwrap();
38/// issuer_params.serial_number = Some(SerialNumber::from(9999));
39/// issuer_params.is_ca = IsCa::Ca(BasicConstraints::Unconstrained);
40/// issuer_params.key_usages = vec![KeyUsagePurpose::KeyCertSign, KeyUsagePurpose::DigitalSignature, KeyUsagePurpose::CrlSign];
41/// #[cfg(feature = "crypto")]
42/// let key_pair = KeyPair::generate().unwrap();
43/// #[cfg(not(feature = "crypto"))]
44/// let key_pair = MyKeyPair { public_key: vec![] };
45/// let issuer = Issuer::new(issuer_params, key_pair);
46///
47/// // Describe a revoked certificate.
48/// let revoked_cert = RevokedCertParams{
49///   serial_number: SerialNumber::from(9999),
50///   revocation_time: date_time_ymd(2024, 06, 17),
51///   reason_code: Some(RevocationReason::KeyCompromise),
52///   invalidity_date: None,
53/// };
54/// // Create a CRL signed by the issuer, revoking revoked_cert.
55/// let crl = CertificateRevocationListParams{
56///   this_update: date_time_ymd(2023, 06, 17),
57///   next_update: date_time_ymd(2024, 06, 17),
58///   crl_number: SerialNumber::from(1234),
59///   issuing_distribution_point: None,
60///   revoked_certs: vec![revoked_cert],
61///   #[cfg(feature = "crypto")]
62///   key_identifier_method: KeyIdMethod::Sha256,
63///   #[cfg(not(feature = "crypto"))]
64///   key_identifier_method: KeyIdMethod::PreSpecified(vec![]),
65/// }.signed_by(&issuer).unwrap();
66///# }
67#[derive(Clone, Debug, PartialEq, Eq)]
68pub struct CertificateRevocationList {
69	der: CertificateRevocationListDer<'static>,
70}
71
72impl CertificateRevocationList {
73	/// Get the CRL in PEM encoded format.
74	#[cfg(feature = "pem")]
75	pub fn pem(&self) -> Result<String, Error> {
76		let p = Pem::new("X509 CRL", &*self.der);
77		Ok(pem::encode_config(&p, ENCODE_CONFIG))
78	}
79
80	/// Get the CRL in DER encoded format.
81	///
82	/// [`CertificateRevocationListDer`] implements `Deref<Target = [u8]>` and `AsRef<[u8]>`,
83	/// so you can easily extract the DER bytes from the return value.
84	pub fn der(&self) -> &CertificateRevocationListDer<'static> {
85		&self.der
86	}
87}

89impl From<CertificateRevocationList> for CertificateRevocationListDer<'static> {
90	fn from(crl: CertificateRevocationList) -> Self {
91		crl.der
92	}

95/// A certificate revocation list (CRL) distribution point, to be included in a certificate's
96/// [distribution points extension](https://www.rfc-editor.org/rfc/rfc5280#section-4.2.1.13) or
97/// a CRL's [issuing distribution point extension](https://datatracker.ietf.org/doc/html/rfc5280#section-5.2.5)
98#[derive(Debug, PartialEq, Eq, Clone)]
99pub struct CrlDistributionPoint {
100	/// One or more URI distribution point names, indicating a place the current CRL can
101	/// be retrieved. When present, SHOULD include at least one LDAP or HTTP URI.
102	pub uris: Vec<String>,
103}
104
105impl CrlDistributionPoint {
106	pub(crate) fn write_der(&self, writer: DERWriter) {
107		// DistributionPoint SEQUENCE
108		writer.write_sequence(|writer| {
109			write_distribution_point_name_uris(writer.next(), &self.uris);
110		});
111	}
112}
113
114fn write_distribution_point_name_uris<'a>(
115	writer: DERWriter,
116	uris: impl IntoIterator<Item = &'a String>,
117) {
118	// distributionPoint DistributionPointName
119	writer.write_tagged_implicit(Tag::context(0), |writer| {
120		writer.write_sequence(|writer| {
121			// fullName GeneralNames
122			writer
123				.next()
124				.write_tagged_implicit(Tag::context(0), |writer| {
125					// GeneralNames
126					writer.write_sequence(|writer| {
127						for uri in uris.into_iter() {
128							// uniformResourceIdentifier [6] IA5String,
129							writer
130								.next()
131								.write_tagged_implicit(Tag::context(6), |writer| {
132									writer.write_ia5_string(uri)
133								});
134						}
135					})
136				});
137		});
138	});
139}

141/// Identifies the reason a certificate was revoked.
142/// See [RFC 5280 §5.3.1][1]
143///
144/// [1]: <https://www.rfc-editor.org/rfc/rfc5280#section-5.3.1>
145#[derive(Debug, Clone, Copy, Eq, PartialEq)]
146#[allow(missing_docs)] // Not much to add above the code name.
147pub enum RevocationReason {
148	Unspecified = 0,
149	KeyCompromise = 1,
150	CaCompromise = 2,
151	AffiliationChanged = 3,
152	Superseded = 4,
153	CessationOfOperation = 5,
154	CertificateHold = 6,
155	// 7 is not defined.
156	RemoveFromCrl = 8,
157	PrivilegeWithdrawn = 9,
158	AaCompromise = 10,
159}

161/// Parameters used for certificate revocation list (CRL) generation
162#[derive(Clone, Debug, PartialEq, Eq)]
163pub struct CertificateRevocationListParams {
164	/// Issue date of the CRL.
165	pub this_update: OffsetDateTime,
166	/// The date by which the next CRL will be issued.
167	pub next_update: OffsetDateTime,
168	/// A monotonically increasing sequence number for a given CRL scope and issuer.
169	pub crl_number: SerialNumber,
170	/// An optional CRL extension identifying the CRL distribution point and scope for a
171	/// particular CRL as described in RFC 5280 Section 5.2.5[^1].
172	///
173	/// [^1]: <https://datatracker.ietf.org/doc/html/rfc5280#section-5.2.5>
174	pub issuing_distribution_point: Option<CrlIssuingDistributionPoint>,
175	/// A list of zero or more parameters describing revoked certificates included in the CRL.
176	pub revoked_certs: Vec<RevokedCertParams>,
177	/// Method to generate key identifiers from public keys
178	///
179	/// Defaults to SHA-256.
180	pub key_identifier_method: KeyIdMethod,
181}
182
183impl CertificateRevocationListParams {
184	/// Serializes the certificate revocation list (CRL).
185	///
186	/// Including a signature from the issuing certificate authority's key.
187	pub fn signed_by(
188		&self,
189		issuer: &Issuer<'_, impl SigningKey>,
190	) -> Result<CertificateRevocationList, Error> {
191		if self.next_update.le(&self.this_update) {
192			return Err(Error::InvalidCrlNextUpdate);
193		}
194
195		if !issuer.key_usages.is_empty() && !issuer.key_usages.contains(&KeyUsagePurpose::CrlSign) {
196			return Err(Error::IssuerNotCrlSigner);
197		}
198
199		// An empty distribution point would be encoded as an empty fullName,
200		// violating GeneralNames ::= SEQUENCE SIZE (1..MAX) OF GeneralName
201		// (RFC 5280 §4.2.1.13).
202		if self
203			.issuing_distribution_point
204			.as_ref()
205			.is_some_and(|idp| idp.distribution_point.uris.is_empty())
206		{
207			return Err(Error::EmptyCrlDistributionPointUris);
208		}
209
210		Ok(CertificateRevocationList {
211			der: self.serialize_der(issuer)?.into(),
212		})
213	}
214
215	fn serialize_der(&self, issuer: &Issuer<'_, impl SigningKey>) -> Result<Vec<u8>, Error> {
216		sign_der(&issuer.signing_key, |writer| {
217			// Write CRL version.
218			// RFC 5280 §5.1.2.1:
219			//   This optional field describes the version of the encoded CRL.  When
220			//   extensions are used, as required by this profile, this field MUST be
221			//   present and MUST specify version 2 (the integer value is 1).
222			// RFC 5280 §5.2:
223			//   Conforming CRL issuers are REQUIRED to include the authority key
224			//   identifier (Section 5.2.1) and the CRL number (Section 5.2.3)
225			//   extensions in all CRLs issued.
226			writer.next().write_u8(1);
227
228			// Write algorithm identifier.
229			// RFC 5280 §5.1.2.2:
230			//   This field MUST contain the same algorithm identifier as the
231			//   signatureAlgorithm field in the sequence CertificateList
232			issuer
233				.signing_key
234				.algorithm()
235				.write_alg_ident(writer.next());
236
237			// Write issuer.
238			// RFC 5280 §5.1.2.3:
239			//   The issuer field MUST contain a non-empty X.500 distinguished name (DN).
240			write_distinguished_name(writer.next(), issuer.distinguished_name.as_ref());
241
242			// Write thisUpdate date.
243			// RFC 5280 §5.1.2.4:
244			//    This field indicates the issue date of this CRL.  thisUpdate may be
245			//    encoded as UTCTime or GeneralizedTime.
246			write_dt_utc_or_generalized(writer.next(), self.this_update);
247
248			// Write nextUpdate date.
249			// While OPTIONAL in the ASN.1 module, RFC 5280 §5.1.2.5 says:
250			//   Conforming CRL issuers MUST include the nextUpdate field in all CRLs.
251			write_dt_utc_or_generalized(writer.next(), self.next_update);
252
253			// Write revokedCertificates.
254			// RFC 5280 §5.1.2.6:
255			//   When there are no revoked certificates, the revoked certificates list
256			//   MUST be absent
257			if !self.revoked_certs.is_empty() {
258				writer.next().write_sequence(|writer| {
259					for revoked_cert in &self.revoked_certs {
260						revoked_cert.write_der(writer.next());
261					}
262				});
263			}
264
265			// Write crlExtensions.
266			// RFC 5280 §5.1.2.7:
267			//   This field may only appear if the version is 2 (Section 5.1.2.1).  If
268			//   present, this field is a sequence of one or more CRL extensions.
269			// RFC 5280 §5.2:
270			//   Conforming CRL issuers are REQUIRED to include the authority key
271			//   identifier (Section 5.2.1) and the CRL number (Section 5.2.3)
272			//   extensions in all CRLs issued.
273			writer.next().write_tagged(Tag::context(0), |writer| {
274				writer.write_sequence(|writer| {
275					// Write authority key identifier.
276					write_x509_authority_key_identifier(
277						writer.next(),
278						self.key_identifier_method
279							.derive(issuer.signing_key.subject_public_key_info()),
280					);
281
282					// Write CRL number.
283					write_x509_extension(writer.next(), oid::CRL_NUMBER, false, |writer| {
284						writer.write_bigint_bytes(self.crl_number.as_ref(), true);
285					});
286
287					// Write issuing distribution point (if present).
288					if let Some(issuing_distribution_point) = &self.issuing_distribution_point {
289						write_x509_extension(
290							writer.next(),
291							oid::CRL_ISSUING_DISTRIBUTION_POINT,
292							true,
293							|writer| {
294								issuing_distribution_point.write_der(writer);
295							},
296						);
297					}
298				});
299			});
300
301			Ok(())
302		})
303	}
304}

306/// A certificate revocation list (CRL) issuing distribution point, to be included in a CRL's
307/// [issuing distribution point extension](https://datatracker.ietf.org/doc/html/rfc5280#section-5.2.5).
308#[derive(Clone, Debug, PartialEq, Eq)]
309pub struct CrlIssuingDistributionPoint {
310	/// The CRL's distribution point, containing a sequence of URIs the CRL can be retrieved from.
311	pub distribution_point: CrlDistributionPoint,
312	/// An optional description of the CRL's scope. If omitted, the CRL may contain
313	/// both user certs and CA certs.
314	pub scope: Option<CrlScope>,
315}
316
317impl CrlIssuingDistributionPoint {
318	fn write_der(&self, writer: DERWriter) {
319		// IssuingDistributionPoint SEQUENCE
320		writer.write_sequence(|writer| {
321			// distributionPoint [0] DistributionPointName OPTIONAL
322			write_distribution_point_name_uris(writer.next(), &self.distribution_point.uris);
323
324			// -- at most one of onlyContainsUserCerts, onlyContainsCACerts,
325			// -- and onlyContainsAttributeCerts may be set to TRUE.
326			if let Some(scope) = self.scope {
327				let tag = match scope {
328					// onlyContainsUserCerts [1] BOOLEAN DEFAULT FALSE,
329					CrlScope::UserCertsOnly => Tag::context(1),
330					// onlyContainsCACerts [2] BOOLEAN DEFAULT FALSE,
331					CrlScope::CaCertsOnly => Tag::context(2),
332				};
333				writer.next().write_tagged_implicit(tag, |writer| {
334					writer.write_bool(true);
335				});
336			}
337		});
338	}
339}

350/// Parameters used for describing a revoked certificate included in a [`CertificateRevocationList`].
351#[derive(Clone, Debug, PartialEq, Eq)]
352pub struct RevokedCertParams {
353	/// Serial number identifying the revoked certificate.
354	pub serial_number: SerialNumber,
355	/// The date at which the CA processed the revocation.
356	pub revocation_time: OffsetDateTime,
357	/// An optional reason code identifying why the certificate was revoked.
358	pub reason_code: Option<RevocationReason>,
359	/// An optional field describing the date on which it was known or suspected that the
360	/// private key was compromised or the certificate otherwise became invalid. This date
361	/// may be earlier than the [`RevokedCertParams::revocation_time`].
362	pub invalidity_date: Option<OffsetDateTime>,
363}
364
365impl RevokedCertParams {
366	fn write_der(&self, writer: DERWriter) {
367		writer.write_sequence(|writer| {
368			// Write serial number.
369			// RFC 5280 §4.1.2.2:
370			//    Certificate users MUST be able to handle serialNumber values up to 20 octets.
371			//    Conforming CAs MUST NOT use serialNumber values longer than 20 octets.
372			//
373			//    Note: Non-conforming CAs may issue certificates with serial numbers
374			//    that are negative or zero.  Certificate users SHOULD be prepared to
375			//    gracefully handle such certificates.
376			writer
377				.next()
378				.write_bigint_bytes(self.serial_number.as_ref(), true);
379
380			// Write revocation date.
381			write_dt_utc_or_generalized(writer.next(), self.revocation_time);
382
383			// Write extensions if applicable.
384			// RFC 5280 §5.3:
385			//   Support for the CRL entry extensions defined in this specification is
386			//   optional for conforming CRL issuers and applications.  However, CRL
387			//   issuers SHOULD include reason codes (Section 5.3.1) and invalidity
388			//   dates (Section 5.3.2) whenever this information is available.
389			// RFC 5280 §5.3.1: "The reason code CRL entry extension SHOULD be
390			// absent instead of using the unspecified (0) reasonCode value."
391			let reason_code = self
392				.reason_code
393				.filter(|reason| *reason != RevocationReason::Unspecified);
394			let has_invalidity_date = self.invalidity_date.is_some();
395			if reason_code.is_some() || has_invalidity_date {
396				writer.next().write_sequence(|writer| {
397					// Write reason code if present.
398					if let Some(reason_code) = reason_code {
399						write_x509_extension(writer.next(), oid::CRL_REASONS, false, |writer| {
400							writer.write_enum(reason_code as i64);
401						});
402					}
403
404					// Write invalidity date if present.
405					// RFC 5280 §5.3.2: InvalidityDate ::= GeneralizedTime.
406					// Unlike the Time CHOICE used elsewhere, dates in the
407					// UTCTime range (1950-2049) must still be encoded as
408					// GeneralizedTime.
409					if let Some(invalidity_date) = self.invalidity_date {
410						write_x509_extension(
411							writer.next(),
412							oid::CRL_INVALIDITY_DATE,
413							false,
414							|writer| {
415								writer.write_generalized_time(&dt_to_generalized(invalidity_date));
416							},
417						)
418					}
419				});
420			}
421		})
422	}

435		let crl = CertificateRevocationListParams {
436			this_update: date_time_ymd(2025, 5, 1),
437			next_update: date_time_ymd(2026, 5, 1),
438			crl_number: SerialNumber::from(1234u64),

441				scope: None,

445		};

447		// A distribution point with no URIs would be encoded as an empty
448		// fullName, violating GeneralNames ::= SEQUENCE SIZE (1..MAX) OF

456	#[test]
457	fn test_unspecified_reason_code_not_written() {
458		let crl = test_crl(RevokedCertParams {
459			serial_number: SerialNumber::from(9999u64),
460			revocation_time: date_time_ymd(2025, 5, 1),
461			reason_code: Some(RevocationReason::Unspecified),
462			invalidity_date: Some(date_time_ymd(2025, 4, 1)),

464
465		let (_rem, parsed) = parse_x509_crl(crl.der()).unwrap();
466		let revoked = parsed.iter_revoked_certificates().next().unwrap();

468		// absent instead of using the unspecified (0) reasonCode value."
469		assert!(revoked

471			.iter()

477		let crl = test_crl(RevokedCertParams {
478			serial_number: SerialNumber::from(9999u64),
479			revocation_time: date_time_ymd(2025, 5, 1),
480			reason_code: None,
481			invalidity_date: Some(date_time_ymd(2025, 4, 1)),

485		let revoked = parsed.iter_revoked_certificates().next().unwrap();
486		assert_eq!(revoked.user_certificate, BigUint::from(9999u64));

489			.extensions()
490			.iter()

494		// CHOICE used elsewhere, dates in the UTCTime range (1950-2049) must
495		// still be encoded as GeneralizedTime.
496		assert_eq!(invalidity_date.value, b"\x18\x0f20250401000000Z");
497	}
