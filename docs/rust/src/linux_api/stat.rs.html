39/// # Safety
40///
41/// `statbuf` must be safe for the kernel to write to.
42pub unsafe fn fstat_raw(fd: core::ffi::c_uint, statbuf: *mut stat) -> Result<(), Errno> {
43    unsafe { syscall!(linux_syscall::SYS_fstat, fd, statbuf) }
44        .check()
45        .map_err(Errno::from)
46}
