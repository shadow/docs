16struct KillOnDrop(Child, StderrForwarder);
17
18impl Drop for KillOnDrop {
19    fn drop(&mut self) {
20        let child = &mut self.0;
21
22        child.kill().ok();
23    }
24}

26fn try_wait_on_child(
27    cmd: &Command,
28    child: &mut Child,
29    stderr_forwarder: &mut StderrForwarder,
30    cargo_output: &CargoOutput,
31) -> Result<Option<()>, Error> {
32    stderr_forwarder.forward_available(cmd);
33
34    match child.try_wait() {
35        Ok(Some(status)) => {
36            stderr_forwarder.forward_all(cmd);
37
38            println!("{status}");
39
40            if status.success() {
41                Ok(Some(()))
42            } else {
43                Err(cargo_output.command_failed(cmd, status))
44            }
45        }
46        Ok(None) => Ok(None),
47        Err(e) => {
48            stderr_forwarder.forward_all(cmd);
49            Err(Error::new(
50                ErrorKind::ToolExecError,
51                format!(
52                    "failed to wait on spawned child process `{}`: {e}",
53                    CommandLine(cmd)
54                ),
55            ))
56        }
57    }
58}

60pub(crate) fn run_commands_in_parallel(
61    cargo_output: &CargoOutput,
62    cmds: &mut dyn Iterator<Item = Result<Command, Error>>,
63) -> Result<(), Error> {
64    // When compiling objects in parallel we do a few dirty tricks to speed
65    // things up:
66    //
67    // * First is that we use the `jobserver` crate to limit the parallelism
68    //   of this build script. The `jobserver` crate will use a jobserver
69    //   configured by Cargo for build scripts to ensure that parallelism is
70    //   coordinated across C compilations and Rust compilations. Before we
71    //   compile anything we make sure to wait until we acquire a token.
72    //
73    //   Note that this jobserver is cached globally so we only used one per
74    //   process and only worry about creating it once.
75    //
76    // * Next we use spawn the process to actually compile objects in
77    //   parallel after we've acquired a token to perform some work
78    //
79    // With all that in mind we compile all objects in a loop here, after we
80    // acquire the appropriate tokens, Once all objects have been compiled
81    // we wait on all the processes and propagate the results of compilation.
82
83    let reactor = Reactor::default();
84
85    // The registrations come before the child, so that they are dropped
86    // first, while what they refer to is still open.
87    let pendings = Cell::new(Vec::<(
88        Command,
89        Option<Registration<'_>>, // stderr_registration
90        ChildRegistration<'_>,
91        KillOnDrop,
92        job_token::JobToken,
93    )>::new());
94    let is_disconnected = Cell::new(false);
95    let has_made_progress = Cell::new(false);
96
97    let wait_future = async {
98        let mut error = None;
99
100        loop {
101            // If the other end of the pipe is already disconnected, then we're not gonna get any new jobs,
102            // so it doesn't make sense to reuse the tokens; in fact,
103            // releasing them as soon as possible (once we know that the other end is disconnected) is beneficial.
104            // Imagine that the last file built takes an hour to finish; in this scenario,
105            // by not releasing the tokens before that last file is done we would effectively block other processes from
106            // starting sooner - even though we only need one token for that last file, not N others that were acquired.
107
108            let pendings_is_empty = cell_modify(&pendings, |pendings| {
109                // Try waiting on them.
110                pendings.retain_mut(|(cmd, stderr_registration, _, child, _token)| {
111                    match try_wait_on_child(cmd, &mut child.0, &mut child.1, cargo_output) {
112                        Ok(Some(())) => {
113                            // Task done, remove the entry
114                            has_made_progress.set(true);
115                            false
116                        }
117                        Ok(None) => {
118                            // Task still not finished, keep the entry. Once
119                            // its stderr is closed there is nothing more to
120                            // wait for on it, and keeping it registered
121                            // would wake every wait at once.
122                            if child.1.stderr().is_none() {
123                                *stderr_registration = None;
124                            }
125                            true
126                        }
127                        Err(err) => {
128                            // Task fail, remove the entry.
129                            // Since we can only return one error, log the error to make
130                            // sure users always see all the compilation failures. The
131                            // logger already got each failed command as `CommandFailed`.
132                            has_made_progress.set(true);
133
134                            if cargo_output.warnings {
135                                println!("cargo:warning={err}");
136                            }
137                            error = Some(err);
138
139                            false
140                        }
141                    }
142                });
143                pendings.is_empty()
144            });
145
146            if pendings_is_empty && is_disconnected.get() {
147                break if let Some(err) = error {
148                    Err(err)
149                } else {
150                    Ok(())
151                };
152            }
153
154            YieldOnce::default().await;
155        }
156    };
157    let spawn_future = async {
158        // Limit our parallelism globally with a jobserver.
159        let mut tokens = job_token::ActiveJobTokenServer::new();
160
161        for res in cmds {
162            let mut cmd = res?;
163            let token = tokens.acquire(&reactor).await?;
164
165            let mut child = spawn(&mut cmd, cargo_output)?;
166            let stderr_forwarder = StderrForwarder::new(&mut child, cargo_output);
167            let mut child = KillOnDrop(child, stderr_forwarder);
168
169            child.1.set_non_blocking()?;
170            let stderr_registration = child
171                .1
172                .stderr()
173                .map(|stderr| reactor.register_stderr(stderr))
174                .transpose()?;
175            let child_registration = reactor.register_child(&child.0)?;
176
177            cell_push(
178                &pendings,
179                (cmd, stderr_registration, child_registration, child, token),
180            );
181
182            has_made_progress.set(true);
183        }
184        is_disconnected.set(true);
185
186        Ok::<_, Error>(())
187    };
188
189    block_on(wait_future, spawn_future, &has_made_progress, &reactor)
190}
