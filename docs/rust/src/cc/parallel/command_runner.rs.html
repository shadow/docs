14struct KillOnDrop(Child, StderrForwarder);

16impl Drop for KillOnDrop {
17    fn drop(&mut self) {
18        let child = &mut self.0;
19
20        child.kill().ok();
21    }
22}

24fn cell_update<T, F>(cell: &Cell<T>, f: F)
25where
26    T: Default,
27    F: FnOnce(T) -> T,
28{
29    let old = cell.take();
30    let new = f(old);
31    cell.set(new);
32}

34fn try_wait_on_child(
35    cmd: &Command,
36    child: &mut Child,
37    stderr_forwarder: &mut StderrForwarder,
38    cargo_output: &CargoOutput,
39) -> Result<Option<()>, Error> {
40    stderr_forwarder.forward_available(cmd);
41
42    match child.try_wait() {
43        Ok(Some(status)) => {
44            stderr_forwarder.forward_all(cmd);
45
46            println!("{status}");
47
48            if status.success() {
49                Ok(Some(()))
50            } else {
51                Err(cargo_output.command_failed(cmd, status))
52            }
53        }
54        Ok(None) => Ok(None),
55        Err(e) => {
56            stderr_forwarder.forward_all(cmd);
57            Err(Error::new(
58                ErrorKind::ToolExecError,
59                format!(
60                    "failed to wait on spawned child process `{}`: {e}",
61                    CommandLine(cmd)
62                ),
63            ))
64        }
65    }
66}

68pub(crate) fn run_commands_in_parallel(
69    cargo_output: &CargoOutput,
70    cmds: &mut dyn Iterator<Item = Result<Command, Error>>,
71) -> Result<(), Error> {
72    // Limit our parallelism globally with a jobserver.
73    let mut tokens = job_token::ActiveJobTokenServer::new();
74
75    // When compiling objects in parallel we do a few dirty tricks to speed
76    // things up:
77    //
78    // * First is that we use the `jobserver` crate to limit the parallelism
79    //   of this build script. The `jobserver` crate will use a jobserver
80    //   configured by Cargo for build scripts to ensure that parallelism is
81    //   coordinated across C compilations and Rust compilations. Before we
82    //   compile anything we make sure to wait until we acquire a token.
83    //
84    //   Note that this jobserver is cached globally so we only used one per
85    //   process and only worry about creating it once.
86    //
87    // * Next we use spawn the process to actually compile objects in
88    //   parallel after we've acquired a token to perform some work
89    //
90    // With all that in mind we compile all objects in a loop here, after we
91    // acquire the appropriate tokens, Once all objects have been compiled
92    // we wait on all the processes and propagate the results of compilation.
93
94    let pendings = Cell::new(Vec::<(Command, KillOnDrop, job_token::JobToken)>::new());
95    let is_disconnected = Cell::new(false);
96    let has_made_progress = Cell::new(false);
97
98    let wait_future = async {
99        let mut error = None;
100
101        loop {
102            // If the other end of the pipe is already disconnected, then we're not gonna get any new jobs,
103            // so it doesn't make sense to reuse the tokens; in fact,
104            // releasing them as soon as possible (once we know that the other end is disconnected) is beneficial.
105            // Imagine that the last file built takes an hour to finish; in this scenario,
106            // by not releasing the tokens before that last file is done we would effectively block other processes from
107            // starting sooner - even though we only need one token for that last file, not N others that were acquired.
108
109            let mut pendings_is_empty = false;
110
111            cell_update(&pendings, |mut pendings| {
112                // Try waiting on them.
113                pendings.retain_mut(|(cmd, child, _token)| {
114                    match try_wait_on_child(cmd, &mut child.0, &mut child.1, cargo_output) {
115                        Ok(Some(())) => {
116                            // Task done, remove the entry
117                            has_made_progress.set(true);
118                            false
119                        }
120                        Ok(None) => true, // Task still not finished, keep the entry
121                        Err(err) => {
122                            // Task fail, remove the entry.
123                            // Since we can only return one error, log the error to make
124                            // sure users always see all the compilation failures. The
125                            // logger already got each failed command as `CommandFailed`.
126                            has_made_progress.set(true);
127
128                            if cargo_output.warnings {
129                                println!("cargo:warning={err}");
130                            }
131                            error = Some(err);
132
133                            false
134                        }
135                    }
136                });
137                pendings_is_empty = pendings.is_empty();
138                pendings
139            });
140
141            if pendings_is_empty && is_disconnected.get() {
142                break if let Some(err) = error {
143                    Err(err)
144                } else {
145                    Ok(())
146                };
147            }
148
149            YieldOnce::default().await;
150        }
151    };
152    let spawn_future = async {
153        for res in cmds {
154            let mut cmd = res?;
155            let token = tokens.acquire().await?;
156            let mut child = spawn(&mut cmd, cargo_output)?;
157            let mut stderr_forwarder = StderrForwarder::new(&mut child, cargo_output);
158            stderr_forwarder.set_non_blocking()?;
159
160            cell_update(&pendings, |mut pendings| {
161                pendings.push((cmd, KillOnDrop(child, stderr_forwarder), token));
162                pendings
163            });
164
165            has_made_progress.set(true);
166        }
167        is_disconnected.set(true);
168
169        Ok::<_, Error>(())
170    };
171
172    block_on(wait_future, spawn_future, &has_made_progress)
173}
