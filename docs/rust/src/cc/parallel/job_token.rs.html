13impl Drop for JobToken {
14    fn drop(&mut self) {
15        match JobTokenServer::new() {
16            JobTokenServer::Inherited(jobserver) => jobserver.release_token_raw(),
17            JobTokenServer::InProcess(jobserver) => jobserver.release_token_raw(),
18        }
19    }
20}

22enum JobTokenServer {
23    Inherited(inherited_jobserver::JobServer),
24    InProcess(inprocess_jobserver::JobServer),
25}

27impl JobTokenServer {
28    /// This function returns a static reference to the jobserver because
29    ///  - creating a jobserver from env is a bit fd-unsafe (e.g. the fd might
30    ///    be closed by other jobserver users in the process) and better do it
31    ///    at the start of the program.
32    ///  - in case a jobserver cannot be created from env (e.g. it's not
33    ///    present), we will create a global in-process only jobserver
34    ///    that has to be static so that it will be shared by all cc
35    ///    compilation.
36    fn new() -> &'static Self {
37        // TODO: Replace with a OnceLock once MSRV is 1.70
38        static JOBSERVER: OnceLock<JobTokenServer> = OnceLock::new();
39
40        JOBSERVER.get_or_init(|| {
41            unsafe { inherited_jobserver::JobServer::from_env() }
42                .map(Self::Inherited)
43                .unwrap_or_else(|| Self::InProcess(inprocess_jobserver::JobServer::new()))
44        })
45    }
46}

48pub(crate) enum ActiveJobTokenServer {
49    Inherited(inherited_jobserver::ActiveJobServer<'static>),
50    InProcess(&'static inprocess_jobserver::JobServer),
51}

53impl ActiveJobTokenServer {
54    pub(crate) fn new() -> Self {
55        match JobTokenServer::new() {
56            JobTokenServer::Inherited(inherited_jobserver) => {
57                Self::Inherited(inherited_jobserver.enter_active())
58            }
59            JobTokenServer::InProcess(inprocess_jobserver) => Self::InProcess(inprocess_jobserver),
60        }
61    }
62
63    pub(crate) async fn acquire(&mut self, reactor: &Reactor) -> Result<JobToken, Error> {
64        match self {
65            Self::Inherited(jobserver) => jobserver.acquire(reactor).await,
66            Self::InProcess(jobserver) => Ok(jobserver.acquire().await),
67        }
68    }
69}

101    pub(super) struct JobServer {
102        /// Implicit token for this process which is obtained and will be
103        /// released in parent. Since `JobTokens` only give back what they got,
104        /// there should be at most one global implicit token in the wild.
105        ///
106        /// Since Rust does not execute any `Drop` for global variables,
107        /// we can't just put it back to jobserver and then re-acquire it at
108        /// the end of the process.
109        ///
110        /// Use `Mutex` to avoid race between acquire and release.
111        /// If an `AtomicBool` is used, then it's possible for:
112        ///  - `release_token_raw`: Tries to set `global_implicit_token` to true, but it is already
113        ///    set  to `true`, continue to release it to jobserver
114        ///  - `acquire` takes the global implicit token, set `global_implicit_token` to false
115        ///  - `release_token_raw` now writes the token back into the jobserver, while
116        ///    `global_implicit_token` is `false`
117        ///
118        /// If the program exits here, then cc effectively increases parallelism by one, which is
119        /// incorrect, hence we use a `Mutex` here.
120        global_implicit_token: Mutex<bool>,
121        inner: jobserver::Client,
122        #[cfg(unix)]
123        fd: Option<OwnedFd>,
124        #[cfg(windows)]
125        semaphore: Option<OwnedHandle>,
126    }
127
128    impl JobServer {
129        pub(super) unsafe fn from_env() -> Option<Self> {
130            jobserver::Client::from_env().map(|inner| Self {
131                #[cfg(unix)]
132                fd: Self::jobserver_fd(&inner),
133                #[cfg(windows)]
134                semaphore: Self::jobserver_semaphore(&inner),
135                inner,
136                global_implicit_token: Mutex::new(true),
137            })
138        }
139
140        /// `jobserver::Client` doesn't expose its pipe, fifo or semaphore, so
141        /// take it from the `MAKEFLAGS` it passes to child processes, which is
142        /// what they use.
143        #[cfg(any(unix, windows))]
144        fn jobserver_auth(client: &jobserver::Client) -> Option<String> {
145            let mut cmd = std::process::Command::new("");
146            client.configure(&mut cmd);
147            let (_, flags) = cmd.get_envs().find(|(key, _)| *key == "CARGO_MAKEFLAGS")?;
148            // Last in the flags, so a fifo path may contain spaces.
149            let (_, auth) = flags?.to_str()?.split_once("--jobserver-auth=")?;
150            Some(auth.to_owned())
151        }
152
153        #[cfg(unix)]
154        fn jobserver_fd(client: &jobserver::Client) -> Option<OwnedFd> {
155            use std::fs::OpenOptions;
156
157            let auth = Self::jobserver_auth(client)?;
158
159            if let Some(path) = auth.strip_prefix("fifo:") {
160                // Read and write, as opening a fifo read-only blocks until it has
161                // a writer.
162                let fifo = OpenOptions::new().read(true).write(true).open(path);
163                Some(fifo.ok()?.into())
164            } else {
165                let (read, _) = auth.split_once(',')?;
166                let read = read.parse::<RawFd>().ok().filter(|fd| *fd >= 0)?;
167                // `from_env` has just opened the client from this fd, and left it
168                // open.
169                unsafe { BorrowedFd::borrow_raw(read) }
170                    .try_clone_to_owned()
171                    .ok()
172            }
173        }
174
175        #[cfg(windows)]
176        fn jobserver_semaphore(client: &jobserver::Client) -> Option<OwnedHandle> {
177            use ::find_msvc_tools::windows_sys::{OpenSemaphoreA, SYNCHRONIZE};
178            use std::{ffi::CString, os::windows::io::FromRawHandle};
179
180            let name = CString::new(Self::jobserver_auth(client)?).ok()?;
181            // Only to wait on, which takes a token, that is then released
182            // through `inner`.
183            let handle = unsafe { OpenSemaphoreA(SYNCHRONIZE, 0, name.as_ptr().cast()) };
184            if handle.is_null() {
185                None
186            } else {
187                Some(unsafe { OwnedHandle::from_raw_handle(handle) })
188            }
189        }
190
191        fn get_global_implicit_token(&self) -> MutexGuard<'_, bool> {
192            self.global_implicit_token
193                .lock()
194                .unwrap_or_else(PoisonError::into_inner)
195        }
196
197        /// All tokens except for the global implicit token will be put back into the jobserver
198        /// immediately and they cannot be cached, since Rust does not call `Drop::drop` on
199        /// global variables.
200        pub(super) fn release_token_raw(&self) {
201            let mut global_implicit_token = self.get_global_implicit_token();
202
203            if *global_implicit_token {
204                // There's already a global implicit token, so this token must
205                // be released back into jobserver.
206                //
207                // `release_raw` should not block
208                let _ = self.inner.release_raw();
209            } else {
210                *global_implicit_token = true;
211            }
212        }
213
214        pub(super) fn enter_active(&self) -> ActiveJobServer<'_> {
215            ActiveJobServer {
216                jobserver: self,
217                helper_thread: None,
218                #[cfg(windows)]
219                semaphore_tokens: Rc::default(),
220            }
221        }
222    }

224    struct HelperThread {
225        inner: jobserver::HelperThread,
226        /// What the helper thread has sent, which only holds it weakly.
227        /// Dropping it puts the tokens left in it back.
228        tokens: Arc<Mutex<Vec<io::Result<jobserver::Acquired>>>>,
229        /// Whether a token was requested and is yet to be received.
230        ///
231        /// Kept across `acquire`s: one that ends up with another token leaves
232        /// its request pending, and requesting again would make the helper
233        /// thread take more tokens than are ever used.
234        has_requested_token: bool,
235        /// Readable once the helper thread has sent a token, to be polled.
236        #[cfg(unix)]
237        notified: File,
238    }
239
240    impl HelperThread {
241        fn new(jobserver: &JobServer) -> Result<Self, Error> {
242            let tokens = Arc::default();
243            let weak = Arc::downgrade(&tokens);
244            #[cfg(unix)]
245            let (notified, notifier) = Self::pipe()?;
246
247            Ok(Self {
248                tokens,
249                inner: jobserver.inner.clone().into_helper_thread(move |res| {
250                    // Only gone if dropping `Self` gave up on joining the
251                    // helper thread, and dropping `res` puts the token back.
252                    let Some(tokens) = weak.upgrade() else {
253                        return;
254                    };
255                    tokens
256                        .lock()
257                        .unwrap_or_else(PoisonError::into_inner)
258                        .push(res);
259                    // Only to wake up the reactor: if it fails, the token is
260                    // still found on its timeout. If the pipe is full, it is
261                    // readable already. If `notified` is gone, it fails with
262                    // `EPIPE`, as Rust programs ignore `SIGPIPE`.
263                    #[cfg(unix)]
264                    let _ = (&notifier).write(&[0]);
265                })?,
266                has_requested_token: false,
267                #[cfg(unix)]
268                notified,
269            })
270        }
271
272        /// A non-blocking pipe, as (read, write).
273        #[cfg(unix)]
274        fn pipe() -> io::Result<(File, File)> {
275            let mut fds = [0; 2];
276            if unsafe { libc::pipe(fds.as_mut_ptr()) } != 0 {
277                return Err(io::Error::last_os_error());
278            }
279            let pipe = unsafe { (File::from_raw_fd(fds[0]), File::from_raw_fd(fds[1])) };
280            for fd in fds {
281                // `pipe2` isn't on all of them.
282                let flags = unsafe { libc::fcntl(fd, libc::F_GETFL) };
283                if flags == -1
284                    || unsafe { libc::fcntl(fd, libc::F_SETFL, flags | libc::O_NONBLOCK) } == -1
285                    || unsafe { libc::fcntl(fd, libc::F_SETFD, libc::FD_CLOEXEC) } == -1
286                {
287                    return Err(io::Error::last_os_error());
288                }
289            }
290            Ok(pipe)
291        }
292
293        fn try_recv(&mut self) -> Result<Option<JobToken>, Error> {
294            // Before receiving, so that a token sent after it notifies again.
295            #[cfg(unix)]
296            loop {
297                match (&self.notified).read(&mut [0; 64]) {
298                    // The helper thread is gone, see below.
299                    Ok(0) => break,
300                    Ok(_) => (),
301                    Err(err) if err.kind() == io::ErrorKind::WouldBlock => break,
302                    Err(err) if err.kind() == io::ErrorKind::Interrupted => (),
303                    // Left readable, it would wake up every wait.
304                    Err(err) => return Err(err.into()),
305                }
306            }
307
308            let mut tokens = self.tokens.lock().unwrap_or_else(PoisonError::into_inner);
309            match tokens.pop() {
310                Some(res) => {
311                    drop(tokens);
312                    self.has_requested_token = false;
313                    res?.drop_without_releasing();
314                    Ok(Some(JobToken::new()))
315                }
316                // The helper thread's callback holds the only `Weak`, and
317                // pushes before dropping it. Read under the lock, so that a
318                // token pushed before is popped above.
319                None if Arc::weak_count(&self.tokens) == 0 => Err(Error::new(
320                    ErrorKind::JobserverHelpThreadError,
321                    "jobserver help thread has returned before ActiveJobServer is dropped",
322                )),
323                None => Ok(None),
324            }
325        }
326
327        /// At most one at a time, see `has_requested_token`.
328        fn request_token(&mut self) {
329            if !self.has_requested_token {
330                self.inner.request_token();
331                self.has_requested_token = true;
332            }
333        }
334    }

336    pub(crate) struct ActiveJobServer<'a> {
337        jobserver: &'a JobServer,
338        helper_thread: Option<HelperThread>,
339        /// Tokens of counts taken from the semaphore by waiting on it.
340        #[cfg(windows)]
341        semaphore_tokens: Rc<Cell<Vec<JobToken>>>,
342    }
343
344    impl ActiveJobServer<'_> {
345        /// Take a token without blocking.
346        fn try_acquire(&self) -> io::Result<Option<JobToken>> {
347            // Already taken by waiting on the semaphore.
348            #[cfg(windows)]
349            if let Some(token) = cell_modify(&self.semaphore_tokens, Vec::pop) {
350                return Ok(Some(token));
351            }
352
353            Ok(self.jobserver.inner.try_acquire()?.map(|acquired| {
354                acquired.drop_without_releasing();
355                JobToken::new()
356            }))
357        }
358
359        pub(super) async fn acquire(&mut self, _reactor: &Reactor) -> Result<JobToken, Error> {
360            // Kept while waiting, dropped once a token is acquired, or the
361            // future is: a jobserver with tokens to spare would wake every
362            // wait at once, and waiting on the semaphore takes tokens.
363            #[cfg(unix)]
364            let mut _registration = None;
365            #[cfg(windows)]
366            let mut _semaphore_registration = None;
367
368            loop {
369                // Fast path
370                if mem::replace(&mut *self.jobserver.get_global_implicit_token(), false) {
371                    break Ok(JobToken::new());
372                }
373
374                match self.try_acquire() {
375                    Ok(Some(token)) => break Ok(token),
376                    Ok(None) => {
377                        // Other processes, such as rustc, other build
378                        // scripts or make, put the tokens they release back
379                        // into the jobserver.
380                        #[cfg(unix)]
381                        if _registration.is_none() {
382                            _registration = self
383                                .jobserver
384                                .fd
385                                .as_ref()
386                                .map(|fd| _reactor.register_fd(fd.as_raw_fd()));
387                        }
388                        #[cfg(windows)]
389                        if _semaphore_registration.is_none() {
390                            if let Some(semaphore) = &self.jobserver.semaphore {
391                                let tokens = Rc::clone(&self.semaphore_tokens);
392                                _semaphore_registration = Some(_reactor.register_semaphore(
393                                    semaphore.as_raw_handle(),
394                                    Box::new(move || cell_push(&tokens, JobToken::new())),
395                                ));
396                            }
397                        }
398                        YieldOnce::default().await
399                    }
400                    Err(err) if err.kind() == io::ErrorKind::Unsupported => {
401                        // Fallback to creating a help thread with blocking acquire
402                        let helper_thread = if let Some(thread) = self.helper_thread.as_mut() {
403                            thread
404                        } else {
405                            self.helper_thread
406                                .insert(HelperThread::new(self.jobserver)?)
407                        };
408
409                        if let Some(token) = helper_thread.try_recv()? {
410                            break Ok(token);
411                        }
412
413                        helper_thread.request_token();
414                        // The helper thread notifies once it has sent a token.
415                        #[cfg(unix)]
416                        if _registration.is_none() {
417                            _registration =
418                                Some(_reactor.register_fd(helper_thread.notified.as_raw_fd()));
419                        }
420                        YieldOnce::default().await
421                    }
422                    Err(err) => break Err(err.into()),
423                }
424            }
425        }
426    }

434    use core::sync::atomic::{
435        AtomicU32,
436        Ordering::{AcqRel, Acquire},
437    };
438
439    pub(crate) struct JobServer(AtomicU32);
440
441    impl JobServer {
442        pub(super) fn new() -> Self {
443            // Use `NUM_JOBS` if set (it's configured by Cargo) and otherwise
444            // just fall back to the number of cores on the local machine, or a reasonable
445            // default if that cannot be determined.
446
447            let parallelism = cargo_env_var_os("NUM_JOBS")
448                .and_then(|j| j.to_str()?.parse::<u32>().ok())
449                .or_else(|| Some(std::thread::available_parallelism().ok()?.get() as u32))
450                .unwrap_or(4);
451
452            Self(AtomicU32::new(parallelism))
453        }
454
455        pub(super) async fn acquire(&self) -> JobToken {
456            loop {
457                // TODO: use try_fetch once msrv bump to 1.95
458                #[allow(deprecated)]
459                let res = self
460                    .0
461                    .fetch_update(AcqRel, Acquire, |tokens| tokens.checked_sub(1));
462
463                if res.is_ok() {
464                    break JobToken::new();
465                }
466
467                YieldOnce::default().await
468            }
469        }
470
471        pub(super) fn release_token_raw(&self) {
472            self.0.fetch_add(1, AcqRel);
473        }
474    }
