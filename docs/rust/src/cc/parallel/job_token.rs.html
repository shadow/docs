13impl Drop for JobToken {
14    fn drop(&mut self) {
15        match JobTokenServer::new() {
16            JobTokenServer::Inherited(jobserver) => jobserver.release_token_raw(),
17            JobTokenServer::InProcess(jobserver) => jobserver.release_token_raw(),
18        }
19    }
20}

22enum JobTokenServer {
23    Inherited(inherited_jobserver::JobServer),
24    InProcess(inprocess_jobserver::JobServer),
25}
26
27impl JobTokenServer {
28    /// This function returns a static reference to the jobserver because
29    ///  - creating a jobserver from env is a bit fd-unsafe (e.g. the fd might
30    ///    be closed by other jobserver users in the process) and better do it
31    ///    at the start of the program.
32    ///  - in case a jobserver cannot be created from env (e.g. it's not
33    ///    present), we will create a global in-process only jobserver
34    ///    that has to be static so that it will be shared by all cc
35    ///    compilation.
36    fn new() -> &'static Self {
37        // TODO: Replace with a OnceLock once MSRV is 1.70
38        static JOBSERVER: OnceLock<JobTokenServer> = OnceLock::new();
39
40        JOBSERVER.get_or_init(|| {
41            unsafe { inherited_jobserver::JobServer::from_env() }
42                .map(Self::Inherited)
43                .unwrap_or_else(|| Self::InProcess(inprocess_jobserver::JobServer::new()))
44        })
45    }
46}

48pub(crate) enum ActiveJobTokenServer {
49    Inherited(inherited_jobserver::ActiveJobServer<'static>),
50    InProcess(&'static inprocess_jobserver::JobServer),
51}
52
53impl ActiveJobTokenServer {
54    pub(crate) fn new() -> Self {
55        match JobTokenServer::new() {
56            JobTokenServer::Inherited(inherited_jobserver) => {
57                Self::Inherited(inherited_jobserver.enter_active())
58            }
59            JobTokenServer::InProcess(inprocess_jobserver) => Self::InProcess(inprocess_jobserver),
60        }
61    }
62
63    pub(crate) async fn acquire(&mut self, reactor: &Reactor) -> Result<JobToken, Error> {
64        match self {
65            Self::Inherited(jobserver) => jobserver.acquire(reactor).await,
66            Self::InProcess(jobserver) => Ok(jobserver.acquire().await),
67        }
68    }
69}

82        io, mem,
83        sync::{mpsc, Mutex, MutexGuard, PoisonError},

86    pub(super) struct JobServer {
87        /// Implicit token for this process which is obtained and will be
88        /// released in parent. Since `JobTokens` only give back what they got,
89        /// there should be at most one global implicit token in the wild.
90        ///
91        /// Since Rust does not execute any `Drop` for global variables,
92        /// we can't just put it back to jobserver and then re-acquire it at
93        /// the end of the process.
94        ///
95        /// Use `Mutex` to avoid race between acquire and release.
96        /// If an `AtomicBool` is used, then it's possible for:
97        ///  - `release_token_raw`: Tries to set `global_implicit_token` to true, but it is already
98        ///    set  to `true`, continue to release it to jobserver
99        ///  - `acquire` takes the global implicit token, set `global_implicit_token` to false
100        ///  - `release_token_raw` now writes the token back into the jobserver, while
101        ///    `global_implicit_token` is `false`
102        ///
103        /// If the program exits here, then cc effectively increases parallelism by one, which is
104        /// incorrect, hence we use a `Mutex` here.
105        global_implicit_token: Mutex<bool>,
106        inner: jobserver::Client,
107        #[cfg(unix)]
108        fd: Option<OwnedFd>,
109    }
110
111    impl JobServer {
112        pub(super) unsafe fn from_env() -> Option<Self> {
113            jobserver::Client::from_env().map(|inner| Self {
114                #[cfg(unix)]
115                fd: Self::jobserver_fd(&inner),
116                inner,
117                global_implicit_token: Mutex::new(true),
118            })
119        }
120
121        /// `jobserver::Client` doesn't expose its pipe or fifo, so take it from
122        /// the `MAKEFLAGS` it passes to child processes, which is what they use.
123        #[cfg(unix)]
124        fn jobserver_fd(client: &jobserver::Client) -> Option<OwnedFd> {
125            use std::{fs::OpenOptions, process::Command};
126
127            let mut cmd = Command::new("");
128            client.configure(&mut cmd);
129            let (_, flags) = cmd.get_envs().find(|(key, _)| *key == "CARGO_MAKEFLAGS")?;
130            // Last in the flags, so a fifo path may contain spaces.
131            let (_, auth) = flags?.to_str()?.split_once("--jobserver-auth=")?;
132
133            if let Some(path) = auth.strip_prefix("fifo:") {
134                // Read and write, as opening a fifo read-only blocks until it has
135                // a writer.
136                let fifo = OpenOptions::new().read(true).write(true).open(path);
137                Some(fifo.ok()?.into())
138            } else {
139                let (read, _) = auth.split_once(',')?;
140                let read = read.parse::<RawFd>().ok().filter(|fd| *fd >= 0)?;
141                // `from_env` has just opened the client from this fd, and left it
142                // open.
143                unsafe { BorrowedFd::borrow_raw(read) }
144                    .try_clone_to_owned()
145                    .ok()
146            }
147        }
148
149        fn get_global_implicit_token(&self) -> MutexGuard<'_, bool> {
150            self.global_implicit_token
151                .lock()
152                .unwrap_or_else(PoisonError::into_inner)
153        }
154
155        /// All tokens except for the global implicit token will be put back into the jobserver
156        /// immediately and they cannot be cached, since Rust does not call `Drop::drop` on
157        /// global variables.
158        pub(super) fn release_token_raw(&self) {
159            let mut global_implicit_token = self.get_global_implicit_token();
160
161            if *global_implicit_token {
162                // There's already a global implicit token, so this token must
163                // be released back into jobserver.
164                //
165                // `release_raw` should not block
166                let _ = self.inner.release_raw();
167            } else {
168                *global_implicit_token = true;
169            }
170        }
171
172        pub(super) fn enter_active(&self) -> ActiveJobServer<'_> {
173            ActiveJobServer {
174                jobserver: self,
175                helper_thread: None,
176            }
177        }
178    }
179
180    struct HelperThread {
181        inner: jobserver::HelperThread,
182        /// When rx is dropped, all the token stored within it will be dropped.
183        rx: mpsc::Receiver<io::Result<jobserver::Acquired>>,
184        /// Whether a token was requested and is yet to be received.
185        ///
186        /// Kept across `acquire`s: one that ends up with another token leaves
187        /// its request pending, and requesting again would make the helper
188        /// thread take more tokens than are ever used.
189        has_requested_token: bool,
190    }
191
192    impl HelperThread {
193        fn new(jobserver: &JobServer) -> Result<Self, Error> {
194            let (tx, rx) = mpsc::channel();
195
196            Ok(Self {
197                rx,
198                inner: jobserver.inner.clone().into_helper_thread(move |res| {
199                    let _ = tx.send(res);
200                })?,
201                has_requested_token: false,
202            })
203        }
204
205        fn try_recv(&mut self) -> Result<Option<JobToken>, Error> {
206            match self.rx.try_recv() {
207                Ok(res) => {
208                    self.has_requested_token = false;
209                    res?.drop_without_releasing();
210                    Ok(Some(JobToken::new()))
211                }
212                Err(mpsc::TryRecvError::Disconnected) => Err(Error::new(
213                    ErrorKind::JobserverHelpThreadError,
214                    "jobserver help thread has returned before ActiveJobServer is dropped",
215                )),
216                Err(mpsc::TryRecvError::Empty) => Ok(None),
217            }
218        }
219
220        /// At most one at a time, see `has_requested_token`.
221        fn request_token(&mut self) {
222            if !self.has_requested_token {
223                self.inner.request_token();
224                self.has_requested_token = true;
225            }
226        }
227    }

229    pub(crate) struct ActiveJobServer<'a> {
230        jobserver: &'a JobServer,
231        helper_thread: Option<HelperThread>,
232    }
233
234    impl ActiveJobServer<'_> {
235        pub(super) async fn acquire(&mut self, _reactor: &Reactor) -> Result<JobToken, Error> {
236            // Kept while waiting, dropped once a token is acquired: a
237            // jobserver with tokens to spare would wake every wait at once.
238            #[cfg(unix)]
239            let mut _registration = None;
240
241            loop {
242                // Fast path
243                if mem::replace(&mut *self.jobserver.get_global_implicit_token(), false) {
244                    break Ok(JobToken::new());
245                }
246
247                match self.jobserver.inner.try_acquire() {
248                    Ok(Some(acquired)) => {
249                        acquired.drop_without_releasing();
250                        break Ok(JobToken::new());
251                    }
252                    Ok(None) => {
253                        // Other processes, such as rustc, other build
254                        // scripts or make, put the tokens they release back
255                        // into the jobserver.
256                        #[cfg(unix)]
257                        if _registration.is_none() {
258                            _registration = self
259                                .jobserver
260                                .fd
261                                .as_ref()
262                                .map(|fd| _reactor.register_fd(fd.as_raw_fd()));
263                        }
264                        YieldOnce::default().await
265                    }
266                    Err(err) if err.kind() == io::ErrorKind::Unsupported => {
267                        // Fallback to creating a help thread with blocking acquire
268                        let helper_thread = if let Some(thread) = self.helper_thread.as_mut() {
269                            thread
270                        } else {
271                            self.helper_thread
272                                .insert(HelperThread::new(self.jobserver)?)
273                        };
274
275                        if let Some(token) = helper_thread.try_recv()? {
276                            break Ok(token);
277                        }
278
279                        helper_thread.request_token();
280                        YieldOnce::default().await
281                    }
282                    Err(err) => break Err(err.into()),
283                }
284            }
285        }
286    }
287}

289mod inprocess_jobserver {
290    use super::JobToken;

292    use crate::{parallel::async_executor::YieldOnce, utilities::cargo_env_var_os};
293
294    use core::sync::atomic::{
295        AtomicU32,
296        Ordering::{AcqRel, Acquire},
297    };
298
299    pub(crate) struct JobServer(AtomicU32);
300
301    impl JobServer {
302        pub(super) fn new() -> Self {
303            // Use `NUM_JOBS` if set (it's configured by Cargo) and otherwise
304            // just fall back to the number of cores on the local machine, or a reasonable
305            // default if that cannot be determined.
306
307            let parallelism = cargo_env_var_os("NUM_JOBS")
308                .and_then(|j| j.to_str()?.parse::<u32>().ok())
309                .or_else(|| Some(std::thread::available_parallelism().ok()?.get() as u32))
310                .unwrap_or(4);
311
312            Self(AtomicU32::new(parallelism))
313        }
314
315        pub(super) async fn acquire(&self) -> JobToken {
316            loop {
317                // TODO: use try_fetch once msrv bump to 1.95
318                #[allow(deprecated)]
319                let res = self
320                    .0
321                    .fetch_update(AcqRel, Acquire, |tokens| tokens.checked_sub(1));
322
323                if res.is_ok() {
324                    break JobToken::new();
325                }
326
327                YieldOnce::default().await
328            }
329        }
330
331        pub(super) fn release_token_raw(&self) {
332            self.0.fetch_add(1, AcqRel);
333        }
334    }
