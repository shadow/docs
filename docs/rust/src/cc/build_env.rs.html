1//! The environment a [`Build`](crate::Build) reads its configuration from and
2//! runs its tools in.
3
4use std::{env, ffi::OsStr, fmt, process::Command, sync::Arc};
5
6use crate::utilities::OnceLock;
7
8/// Environment variables, in the order they are applied.
9pub(crate) type EnvVars = [(Arc<OsStr>, Arc<OsStr>)];
10
11/// The environment of a [`Build`](crate::Build): a snapshot of the process
12/// environment, taken on the first read and used for cc's own lookups, then
13/// the entries set with [`Build::env`](crate::Build::env), which win over it in
14/// child processes.
15#[derive(Clone, Debug, Default)]
16pub(crate) struct BuildEnv {
17    /// Private, so that nothing can read the environment without taking the
18    /// snapshot first.
19    inherited: OnceLock<EnvSnapshot>,
20    pub(crate) explicit: Vec<(Arc<OsStr>, Arc<OsStr>)>,
21}
22
23impl BuildEnv {
24    /// An environment whose snapshot is already taken.
25    pub(crate) fn with_inherited(
26        inherited: EnvSnapshot,
27        explicit: Vec<(Arc<OsStr>, Arc<OsStr>)>,
28    ) -> Self {
29        let cell = OnceLock::new();
30        cell.get_or_init(|| inherited);
31        Self {
32            inherited: cell,
33            explicit,
34        }
35    }
36
37    /// The process environment as it was when this was first called.
38    pub(crate) fn inherited(&self) -> &EnvSnapshot {
39        self.inherited.get_or_init(EnvSnapshot::capture)
40    }
41
42    /// Make `cmd` run in this environment. Like [`EnvSnapshot::apply`], this
43    /// clears anything set on `cmd`'s environment before.
44    pub(crate) fn apply(&self, cmd: &mut Command) {
45        self.inherited().apply(cmd);
46        cmd.envs(pairs(&self.explicit));
47    }
48}
49
50/// A copy of the process environment. Clones share the copy.
51#[derive(Clone, PartialEq, Eq, PartialOrd, Ord, Hash)]
52pub(crate) struct EnvSnapshot {
53    vars: Arc<EnvVars>,
54}
55
56impl EnvSnapshot {
57    /// Copy the process environment as it is now.
58    pub(crate) fn capture() -> Self {
59        Self {
60            vars: env::vars_os()
61                .map(|(key, value)| (key.into(), value.into()))
62                .collect(),
63        }
64    }
65
66    #[cfg(test)]
67    pub(crate) fn from_pairs(vars: &[(&str, &str)]) -> Self {
68        Self {
69            vars: vars
70                .iter()
71                .map(|(key, value)| (OsStr::new(key).into(), OsStr::new(value).into()))
72                .collect(),
73        }
74    }
75
76    /// Look up `key` the way [`env::var_os`] would have when the snapshot was
77    /// taken.
78    pub(crate) fn get(&self, key: &OsStr) -> Option<&Arc<OsStr>> {
79        self.vars
80            .iter()
81            .find(|(k, _)| is_same_key(k, key))
82            .map(|(_, value)| value)
83    }
84
85    /// Make `cmd` run in this environment rather than the current one.
86    ///
87    /// This clears the environment `cmd` would inherit and sets every variable
88    /// of the snapshot, so the child does not depend on the process environment
89    /// at the time it is spawned, which another thread may be changing. It also
90    /// clears anything set on `cmd`'s environment before, so call it first.
91    pub(crate) fn apply(&self, cmd: &mut Command) {
92        cmd.env_clear();
93        cmd.envs(pairs(&self.vars));
94    }
95}

97impl fmt::Debug for EnvSnapshot {
98    // The values may hold secrets, and a `Build` can end up in a log.
99    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
100        write!(f, "EnvSnapshot(<{} variables>)", self.vars.len())
101    }
102}
103
104fn pairs(vars: &EnvVars) -> impl Iterator<Item = (&OsStr, &OsStr)> {
105    vars.iter().map(|(key, value)| (&**key, &**value))
106}
107
108/// Environment variable names are case-insensitive on Windows.
109fn is_same_key(a: &OsStr, b: &OsStr) -> bool {
110    if cfg!(windows) {
111        a.eq_ignore_ascii_case(b)
112    } else {
113        a == b
114    }
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120    use std::ffi::OsString;
121
122    fn snapshot(vars: &[(&str, &str)]) -> EnvSnapshot {
123        EnvSnapshot::from_pairs(vars)
124    }
125
126    fn envs(cmd: &Command) -> Vec<(OsString, Option<OsString>)> {
127        cmd.get_envs()
128            .map(|(key, value)| (key.to_owned(), value.map(OsStr::to_owned)))
129            .collect()
130    }
131
132    #[test]
133    fn apply_replaces_the_whole_environment() {
134        let mut cmd = Command::new("cc");
135        cmd.env("CC_TEST_SET_BEFORE", "1")
136            .env_remove("CC_TEST_REMOVED_BEFORE");
137        snapshot(&[("CC_TEST_IN_SNAPSHOT", "1")]).apply(&mut cmd);
138        assert_eq!(
139            envs(&cmd),
140            [("CC_TEST_IN_SNAPSHOT".into(), Some("1".into()))]
141        );
142    }
143
144    #[test]
145    fn build_env_wins_over_snapshot() {
146        let env = BuildEnv::with_inherited(
147            snapshot(&[("CC_TEST_ORDER", "inherited")]),
148            vec![(
149                OsStr::new("CC_TEST_ORDER").into(),
150                OsStr::new("explicit").into(),
151            )],
152        );
153        let mut cmd = Command::new("cc");
154        env.apply(&mut cmd);
155        assert_eq!(
156            envs(&cmd),
157            [("CC_TEST_ORDER".into(), Some("explicit".into()))]
158        );
159    }
160
161    #[test]
162    fn snapshot_lookup_shares_the_value() {
163        let snapshot = snapshot(&[("CC_TEST_SHARED", "a")]);
164        let value = snapshot.get(OsStr::new("CC_TEST_SHARED")).unwrap();
165        assert!(Arc::ptr_eq(value, &snapshot.vars[0].1));
166    }
167
168    #[test]
169    fn snapshot_lookup_follows_the_platform_case_rule() {
170        let snapshot = snapshot(&[("Path", "a")]);
171        assert_eq!(
172            snapshot.get(OsStr::new("Path")).map(|value| &**value),
173            Some(OsStr::new("a"))
174        );
175        let other_case = snapshot.get(OsStr::new("PATH")).map(|value| &**value);
176        if cfg!(windows) {
177            assert_eq!(other_case, Some(OsStr::new("a")));
178        } else {
179            assert_eq!(other_case, None);
180        }
181    }
182
183    #[test]
184    fn snapshot_debug_hides_values() {
185        let debug = format!("{:?}", snapshot(&[("CC_TEST_SECRET", "hunter2")]));
186        assert!(!debug.contains("hunter2"), "{debug}");
187    }
