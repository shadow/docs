13/// Receives cc's messages, set with
14/// [`Build::message_logger`](crate::Build::message_logger).
15///
16/// cc calls the logger on the thread that runs the `Build` method, also with
17/// the `parallel` feature. Clones of a `Build` share the logger and may run on
18/// other threads, so any synchronisation it needs is up to the logger.
19pub trait BuildMessageLogger: Send + Sync + 'static {
20    /// Called for each message.
21    ///
22    /// `kind` says what the message is about, `msg` holds its text, and
23    /// `extra` holds more about it, depending on `kind`:
24    ///
25    /// - [`GeneralWarning`](BuildMessageKind::GeneralWarning): `()`
26    /// - [`StderrForwarding`](BuildMessageKind::StderrForwarding) and
27    ///   [`CommandFailed`](BuildMessageKind::CommandFailed): the
28    ///   [`Command`](std::process::Command) that was run, read with
29    ///   `extra.downcast_ref::<Command>()`
30    /// - [`CompileCommand`](BuildMessageKind::CompileCommand): the
31    ///   [`CompileCommand`]
32    ///
33    /// The logger can't fail the build, but a panic in it reaches the caller
34    /// of the `Build` method that was running.
35    fn log(&self, kind: BuildMessageKind, msg: BuildMessage<'_>, extra: &dyn Any);
36}
37
38/// What a [`BuildMessage`] is about.
39#[derive(Clone, Copy, Debug, PartialEq, Eq)]
40#[non_exhaustive]
41pub enum BuildMessageKind {
42    /// Every warning cc prints itself, such as a failed compiler family
43    /// detection or a setting that doesn't apply to the compiler. A failed
44    /// command's error, which `parallel` also prints as a warning, arrives as
45    /// [`CommandFailed`](BuildMessageKind::CommandFailed) instead.
46    ///
47    /// `extra` is `()`.
48    GeneralWarning,
49    /// Each line of stderr that cc forwards from a command it runs, such as
50    /// the compiler or the archiver. Stderr that cc hides, such as that of
51    /// flag support probes, isn't passed.
52    ///
53    /// A trailing `\r` is removed, and text that isn't UTF-8 is converted
54    /// lossily. `extra` is the [`Command`](std::process::Command).
55    StderrForwarding,
56    /// A command cc runs exits unsuccessfully.
57    ///
58    /// The text is the error cc returns, and `extra` is the
59    /// [`Command`](std::process::Command). Checks whose failure cc expects,
60    /// such as flag support probes and the `ar` call with `D` that cc retries
61    /// without it, are not reported. Neither are commands that can't be
62    /// started, since they have no exit status.
63    #[non_exhaustive]
64    CommandFailed {
65        /// `true` for detection commands that don't break the build: cc
66        /// recovers when they fail, so the build can still succeed. Examples
67        /// are the compiler's `--print-search-dirs` query,
68        /// `xcrun --show-sdk-version` and compiler family detection. cc hides
69        /// the output of family detection unless
70        /// [`cargo_debug`](crate::Build::cargo_debug) is on, so its failures
71        /// are only reported then, apart from the `--version` check it runs on
72        /// clang. A failed family detection still arrives as a
73        /// [`GeneralWarning`](BuildMessageKind::GeneralWarning).
74        ///
75        /// `false` for commands whose failure fails the build, such as the
76        /// compiler, the archiver or `xcrun --show-sdk-path`.
77        is_detection_cmd: bool,
78        /// The status the command exited with.
79        exit_status: ExitStatus,
80    },
81    /// cc is about to run the command that compiles one source file into an
82    /// object file, during [`Build::compile`](crate::Build::compile) or
83    /// [`Build::compile_intermediates`](crate::Build::compile_intermediates).
84    /// Each object file gets one message, also with the `parallel` feature.
85    ///
86    /// The text is the command line, and `extra` is the
87    /// [`CompileCommand`],
88    /// which a
89    /// [`CompileCommandCollector`](crate::compile_commands::CompileCommandCollector)
90    /// collects to write a `compile_commands.json`.
91    #[cfg_attr(
92        not(feature = "compile-commands"),
93        allow(rustdoc::broken_intra_doc_links)
94    )]
95    CompileCommand,
96}

98/// The command that compiles one object file: an entry of a
99/// [JSON compilation database](https://clang.llvm.org/docs/JSONCompilationDatabase.html).
100///
101/// cc passes it as the `extra` of a [`BuildMessageKind::CompileCommand`]
102/// message.
103#[derive(Clone, Debug, PartialEq, Eq)]
104pub struct CompileCommand {
105    pub(crate) directory: Box<Path>,
106    pub(crate) file: Box<Path>,
107    pub(crate) arguments: Box<[Box<OsStr>]>,
108    pub(crate) output: Box<Path>,
109}
110
111impl CompileCommand {
112    /// `cmd` compiles `src` into `dst` and runs in `directory`, which must be
113    /// absolute.
114    pub(crate) fn new(cmd: &Command, src: &Path, dst: &Path, directory: &Path) -> Self {
115        // cc runs its compile commands in its own working directory.
116        debug_assert!(cmd.get_current_dir().is_none());
117        Self {
118            directory: directory.into(),
119            file: directory.join(src).into(),
120            arguments: std::iter::once(cmd.get_program())
121                .chain(cmd.get_args())
122                .map(Box::from)
123                .collect(),
124            output: directory.join(dst).into(),
125        }
126    }
127
128    /// The working directory of the compiler, an absolute path. Relative paths
129    /// in the arguments are relative to it.
130    pub fn directory(&self) -> &Path {
131        &self.directory
132    }
133
134    /// The source file, an absolute path.
135    pub fn file(&self) -> &Path {
136        &self.file
137    }
138
139    /// The program cc runs, followed by its arguments, without any escaping.
140    pub fn arguments(&self) -> impl ExactSizeIterator<Item = &OsStr> + DoubleEndedIterator {
141        self.arguments.iter().map(|arg| &**arg)
142    }
143
144    /// The object file, an absolute path.
145    pub fn output(&self) -> &Path {
146        &self.output
147    }
148}

150/// The text of a message passed to a [`BuildMessageLogger`].
151#[derive(Clone, Copy, Debug)]
152#[non_exhaustive]
153pub enum BuildMessage<'a> {
154    /// The message as a string.
155    Str(&'a str),
156}
157
158impl fmt::Display for BuildMessage<'_> {
159    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
160        match self {
161            Self::Str(s) => f.write_str(s),
162        }
163    }
164}

166/// The logger of a `Build`, shared by its clones, or the boxed one a
167/// [`CompileCommandCollector`](crate::compile_commands::CompileCommandCollector)
168/// forwards to.
169#[derive(Clone)]
170pub(crate) struct Logger<L = Arc<dyn BuildMessageLogger>>(pub(crate) L);
171
172// cc only calls the logger and never relies on its state, so a panic can't
173// leave cc with a broken one. This keeps `Build` and `CompileCommandCollector`
174// `UnwindSafe` and `RefUnwindSafe`.
175impl<L> UnwindSafe for Logger<L> {}
176impl<L> RefUnwindSafe for Logger<L> {}
177
178impl Logger {
179    pub(crate) fn log(&self, kind: BuildMessageKind, msg: &str, extra: &dyn Any) {
180        self.0.log(kind, BuildMessage::Str(msg), extra);
181    }
182}
183
184impl<L> fmt::Debug for Logger<L> {
185    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
186        f.debug_struct("Logger").finish_non_exhaustive()
187    }
188}

190#[cfg(test)]
191mod tests {
192    use super::*;
193
194    #[test]
195    fn compile_command_has_no_spare_capacity() {
196        // Each field is a boxed slice, a pointer and a length, without the
197        // capacity a `PathBuf` or `Vec` would add.
198        assert_eq!(
199            std::mem::size_of::<CompileCommand>(),
200            4 * std::mem::size_of::<Box<[u8]>>()
201        );
202    }
203
204    #[test]
205    fn paths_are_joined_to_the_directory() {
206        let directory = Path::new(if cfg!(windows) { r"C:\crate" } else { "/crate" });
207        let mut cmd = Command::new("cc");
208        cmd.args(["-c", "src/foo.c", "-o", "out/foo.o"]);
209
210        let command = CompileCommand::new(
211            &cmd,
212            Path::new("src/foo.c"),
213            Path::new("out/foo.o"),
214            directory,
215        );
216        assert_eq!(command.directory(), directory);
217        assert_eq!(command.file(), directory.join("src/foo.c"));
218        assert_eq!(command.output(), directory.join("out/foo.o"));
219        assert!(command
220            .arguments()
221            .eq(["cc", "-c", "src/foo.c", "-o", "out/foo.o"]));
222
223        // An absolute source stays as it is.
224        let src = directory.join("other").join("bar.c");
225        let command = CompileCommand::new(&cmd, &src, Path::new("out/bar.o"), directory);
226        assert_eq!(command.file(), src);
227    }
