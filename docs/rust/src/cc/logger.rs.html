11/// Receives cc's messages, set with
12/// [`Build::message_logger`](crate::Build::message_logger).
13///
14/// cc calls the logger on the thread that runs the `Build` method, also with
15/// the `parallel` feature. Clones of a `Build` share the logger and may run on
16/// other threads, so any synchronisation it needs is up to the logger.
17pub trait BuildMessageLogger: Send + Sync + 'static {
18    /// Called for each message.
19    ///
20    /// `kind` says what the message is about, `msg` holds its text, and
21    /// `extra` holds more about it, depending on `kind`:
22    ///
23    /// - [`GeneralWarning`](BuildMessageKind::GeneralWarning): `()`
24    /// - [`StderrForwarding`](BuildMessageKind::StderrForwarding) and
25    ///   [`CommandFailed`](BuildMessageKind::CommandFailed): the
26    ///   [`Command`](std::process::Command) that was run, read with
27    ///   `extra.downcast_ref::<Command>()`
28    ///
29    /// The logger can't fail the build, but a panic in it reaches the caller
30    /// of the `Build` method that was running.
31    fn log(&self, kind: BuildMessageKind, msg: BuildMessage<'_>, extra: &dyn Any);
32}
33
34/// What a [`BuildMessage`] is about.
35#[derive(Clone, Copy, Debug, PartialEq, Eq)]
36#[non_exhaustive]
37pub enum BuildMessageKind {
38    /// Every warning cc prints itself, such as a failed compiler family
39    /// detection or a setting that doesn't apply to the compiler. A failed
40    /// command's error, which `parallel` also prints as a warning, arrives as
41    /// [`CommandFailed`](BuildMessageKind::CommandFailed) instead.
42    ///
43    /// `extra` is `()`.
44    GeneralWarning,
45    /// Each line of stderr that cc forwards from a command it runs, such as
46    /// the compiler or the archiver. Stderr that cc hides, such as that of
47    /// flag support probes, isn't passed.
48    ///
49    /// A trailing `\r` is removed, and text that isn't UTF-8 is converted
50    /// lossily. `extra` is the [`Command`](std::process::Command).
51    StderrForwarding,
52    /// A command cc runs exits unsuccessfully.
53    ///
54    /// The text is the error cc returns, and `extra` is the
55    /// [`Command`](std::process::Command). Checks whose failure cc expects,
56    /// such as flag support probes and the `ar` call with `D` that cc retries
57    /// without it, are not reported. Neither are commands that can't be
58    /// started, since they have no exit status.
59    #[non_exhaustive]
60    CommandFailed {
61        /// `true` for detection commands that don't break the build: cc
62        /// recovers when they fail, so the build can still succeed. Examples
63        /// are the compiler's `--print-search-dirs` query,
64        /// `xcrun --show-sdk-version` and compiler family detection. cc hides
65        /// the output of family detection unless
66        /// [`cargo_debug`](crate::Build::cargo_debug) is on, so its failures
67        /// are only reported then, apart from the `--version` check it runs on
68        /// clang. A failed family detection still arrives as a
69        /// [`GeneralWarning`](BuildMessageKind::GeneralWarning).
70        ///
71        /// `false` for commands whose failure fails the build, such as the
72        /// compiler, the archiver or `xcrun --show-sdk-path`.
73        is_detection_cmd: bool,
74        /// The status the command exited with.
75        exit_status: ExitStatus,
76    },
77}
78
79/// The text of a message passed to a [`BuildMessageLogger`].
80#[derive(Clone, Copy, Debug)]
81#[non_exhaustive]
82pub enum BuildMessage<'a> {
83    /// The message as a string.
84    Str(&'a str),
85}
86
87impl fmt::Display for BuildMessage<'_> {
88    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
89        match self {
90            Self::Str(s) => f.write_str(s),
91        }
92    }
93}
94
95/// The logger of a `Build`, shared by its clones.
96#[derive(Clone)]
97pub(crate) struct Logger(pub(crate) Arc<dyn BuildMessageLogger>);
98
99// cc only calls the logger and never relies on its state, so a panic can't
100// leave cc with a broken one. This keeps `Build` `UnwindSafe` and
101// `RefUnwindSafe`.
102impl UnwindSafe for Logger {}
103impl RefUnwindSafe for Logger {}
104
105impl Logger {
106    pub(crate) fn log(&self, kind: BuildMessageKind, msg: &str, extra: &dyn Any) {
107        self.0.log(kind, BuildMessage::Str(msg), extra);
108    }
109}
110
111impl fmt::Debug for Logger {
112    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
113        f.debug_struct("Logger").finish_non_exhaustive()
114    }
